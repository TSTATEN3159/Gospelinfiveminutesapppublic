a) Progress helper (localStorage)
Create a small helper file:
// client/src/features/discipleship/discipleshipProgress.ts

import { DiscipleshipPlan } from "./discipleshipPlans";

const PROGRESS_PREFIX = "discipleship-progress:";

export type PlanProgress = {
  completedItemIds: string[];
};

function getStorageKey(planId: string) {
  return `${PROGRESS_PREFIX}${planId}`;
}

export function loadPlanProgress(plan: DiscipleshipPlan): {
  completed: number;
  total: number;
  ratio: number;
  completedItemIds: string[];
} {
  const key = getStorageKey(plan.id);
  let stored: PlanProgress | null = null;

  try {
    const raw = localStorage.getItem(key);
    if (raw) stored = JSON.parse(raw);
  } catch {
    stored = null;
  }

  const completedIds = stored?.completedItemIds ?? [];
  const allItemIds = plan.days.flatMap((d) => d.items.map((i) => i.id));

  const completed = allItemIds.filter((id) => completedIds.includes(id)).length;
  const total = allItemIds.length;
  const ratio = total === 0 ? 0 : completed / total;

  return { completed, total, ratio, completedItemIds: completedIds };
}

export function markItemCompleted(plan: DiscipleshipPlan, itemId: string) {
  const key = getStorageKey(plan.id);
  const current = loadPlanProgress(plan);
  if (current.completedItemIds.includes(itemId)) return;

  const updated: PlanProgress = {
    completedItemIds: [...current.completedItemIds, itemId],
  };

  localStorage.setItem(key, JSON.stringify(updated));
}
b) Show progress bar on the Discipleship list tiles
Update DiscipleshipScreen.tsx:
// client/src/features/discipleship/DiscipleshipScreen.tsx
import { Link } from "react-router-dom";
import { DISCIPLESHIP_PLANS } from "./discipleshipPlans";
import { loadPlanProgress } from "./discipleshipProgress";

export default function DiscipleshipScreen() {
  return (
    <div className="flex flex-col h-full bg-slate-50">
      <header className="px-4 pt-4 pb-2">
        <h1 className="text-2xl font-semibold tracking-tight">Discipleship</h1>
        <p className="text-sm text-slate-500">
          Short Bible plans to help you follow Jesus in everyday life.
        </p>
      </header>

      <main className="flex-1 overflow-y-auto px-4 pb-6 space-y-4">
        {DISCIPLESHIP_PLANS.map((plan) => {
          const progress = loadPlanProgress(plan);
          const percent = Math.round(progress.ratio * 100);

          return (
            <Link
              key={plan.id}
              to={`/discipleship/plans/${plan.id}`}
              className="flex rounded-2xl bg-white shadow-sm overflow-hidden hover:shadow-md transition-shadow"
            >
              <div className="w-24 h-24 flex-shrink-0">
                <img
                  src={plan.imageUrl}
                  alt={plan.title}
                  className="w-full h-full object-cover"
                />
              </div>

              <div className="flex-1 p-3 flex flex-col justify-between">
                <div>
                  <h2 className="text-base font-semibold leading-snug">
                    {plan.title}
                  </h2>
                  {plan.subtitle && (
                    <p className="text-xs text-slate-500 mt-0.5">
                      {plan.subtitle}
                    </p>
                  )}
                </div>

                {/* Progress meter */}
                <div className="mt-2">
                  <div className="h-1.5 w-full bg-slate-200 rounded-full overflow-hidden">
                    <div
                      className="h-full bg-slate-900 transition-all"
                      style={{ width: `${percent}%` }}
                    />
                  </div>
                  <p className="text-[11px] text-slate-500 mt-1">
                    {progress.completed}/{progress.total} steps • {percent}% done
                  </p>
                </div>
              </div>
            </Link>
          );
        })}
      </main>
    </div>
  );
}
