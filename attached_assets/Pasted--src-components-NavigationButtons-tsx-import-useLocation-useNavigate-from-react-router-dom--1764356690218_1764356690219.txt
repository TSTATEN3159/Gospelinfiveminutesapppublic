// src/components/NavigationButtons.tsx
import { useLocation, useNavigate } from "react-router-dom";
import { appRoutes } from "@/routesConfig";

export default function NavigationButtons() {
  const location = useLocation();
  const navigate = useNavigate();

  // Find current route in the config by path
  const currentIndex = appRoutes.findIndex(
    (route) => route.path === location.pathname
  );

  const hasPrev = currentIndex > 0;
  const hasNext = currentIndex >= 0 && currentIndex < appRoutes.length - 1;

  const prevRoute = hasPrev ? appRoutes[currentIndex - 1] : null;
  const nextRoute = hasNext ? appRoutes[currentIndex + 1] : null;
  const homeRoute = appRoutes[0];

  const handleGoBack = () => {
    if (hasPrev && prevRoute) {
      navigate(prevRoute.path);
    } else {
      // If no "previous" in config, just go back in history
      navigate(-1);
    }
  };

  const handleGoHome = () => {
    navigate(homeRoute.path);
  };

  const handleGoNext = () => {
    if (hasNext && nextRoute) {
      navigate(nextRoute.path);
    }
  };

  // If we couldn't find current route in config, show just a Home button
  if (currentIndex === -1) {
    return (
      <div className="w-full border-t border-slate-800/50 mt-4 pt-3 flex justify-center gap-3">
        <button
          onClick={handleGoHome}
          className="px-4 py-2 rounded-full text-sm font-medium border border-slate-500"
        >
          Home
        </button>
      </div>
    );
  }

  return (
    <div className="w-full border-t border-slate-800/50 mt-4 pt-3 flex justify-between items-center gap-2 text-xs sm:text-sm">
      <button
        onClick={handleGoBack}
        disabled={!hasPrev}
        className={`px-3 py-2 rounded-full border ${
          hasPrev
            ? "border-slate-500"
            : "border-slate-700 text-slate-500 opacity-50 cursor-not-allowed"
        }`}
      >
        ◀ {hasPrev && prevRoute ? prevRoute.label : "Back"}
      </button>

      <button
        onClick={handleGoHome}
        className="px-4 py-2 rounded-full border border-slate-300 font-semibold"
      >
        Home
      </button>

      <button
        onClick={handleGoNext}
        disabled={!hasNext}
        className={`px-3 py-2 rounded-full border ${
          hasNext
            ? "border-slate-500"
            : "border-slate-700 text-slate-500 opacity-50 cursor-not-allowed"
        }`}
      >
        {hasNext && nextRoute ? nextRoute.label : "Next"} ▶
      </button>
    </div>
  );
}
