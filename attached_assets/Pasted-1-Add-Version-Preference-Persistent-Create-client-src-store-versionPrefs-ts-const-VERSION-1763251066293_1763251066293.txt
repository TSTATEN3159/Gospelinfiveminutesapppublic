1️⃣ Add Version Preference (Persistent)

📄 Create → client/src/store/versionPrefs.ts

const VERSION_KEY = "preferredBibleVersion";

export function getVersion(): string {
  return localStorage.getItem(VERSION_KEY) || "KJV";
}

export function setVersion(version: string) {
  localStorage.setItem(VERSION_KEY, version);
}

5️⃣ Reusable ScriptureCard With Tap-to-Read

📄 Create → client/src/components/ScriptureCard.tsx

import { toggleSpeech } from "@/utils/speechEngine";

export default function ScriptureCard({ text, version }: { text: string; version: string }) {
  if (!text) return null;

  return (
    <div
      className="rounded-lg border p-4 bg-white dark:bg-gray-800"
      onClick={() => toggleSpeech(text)}
    >
      <div className="text-xs text-gray-500 mb-1">{version}</div>
      <div className="text-lg leading-relaxed">{text}</div>
    </div>
  );
}
