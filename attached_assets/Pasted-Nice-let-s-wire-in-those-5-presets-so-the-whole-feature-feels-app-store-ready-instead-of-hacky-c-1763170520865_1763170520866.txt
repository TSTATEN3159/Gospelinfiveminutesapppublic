3️⃣ Add preset application logic
Inside ScriptureImageGenerator (in the component body, near the other handlers like handleFileUpload, handleGeneratePreview, etc.), add:
  const applyPreset = (preset: ScripturePreset) => {
    setSettings((prev) => ({
      ...prev,
      ...preset.overrides,
      // Keep verse, reference, and background as the user already chose
      verseText: prev.verseText,
      verseReference: prev.verseReference,
      backgroundId: prev.backgroundId,
    }));
    toast({
      title: `${preset.name} preset applied`,
      description: 'You can still fine-tune settings in the Customize tab.',
    });
  };
