3️⃣ Plan detail screen (days + items)
Create: client/src/features/discipleship/DiscipleshipPlanScreen.tsx
// client/src/features/discipleship/DiscipleshipPlanScreen.tsx
import { useMemo, useState } from "react";
import { useNavigate, useParams, Link } from "react-router-dom";
import { DISCIPLESHIP_PLANS, DiscipleshipPlan, PlanDay } from "./discipleshipPlans";

export default function DiscipleshipPlanScreen() {
  const { planId } = useParams<{ planId: string }>();
  const navigate = useNavigate();

  const plan: DiscipleshipPlan | undefined = useMemo(
    () => DISCIPLESHIP_PLANS.find((p) => p.id === planId),
    [planId]
  );

  const [activeDayIndex, setActiveDayIndex] = useState(0);

  if (!plan) {
    return (
      <div className="p-4">
        <p className="text-sm text-red-600">Plan not found.</p>
        <Link to="/discipleship" className="text-blue-600 text-sm mt-2 inline-block">
          Back to Discipleship
        </Link>
      </div>
    );
  }

  const activeDay: PlanDay = plan.days[activeDayIndex];

  return (
    <div className="flex flex-col h-full bg-slate-50">
      {/* Top bar */}
      <div className="flex items-center px-4 pt-4 pb-2">
        <button
          onClick={() => navigate(-1)}
          className="mr-3 text-sm text-slate-600"
        >
          ‹ Back
        </button>
        <h1 className="text-lg font-semibold truncate">{plan.title}</h1>
      </div>

      {/* Header image */}
      <div className="px-4">
        <div className="rounded-3xl overflow-hidden shadow-md mb-3">
          <img
            src={plan.imageUrl}
            alt={plan.title}
            className="w-full h-44 object-cover"
          />
        </div>
      </div>

      {/* Day selector */}
      <div className="px-4 flex space-x-2 mb-2">
        {plan.days.map((day, index) => {
          const isActive = index === activeDayIndex;
          return (
            <button
              key={day.id}
              onClick={() => setActiveDayIndex(index)}
              className={`flex-1 rounded-2xl border text-center py-2 text-xs ${
                isActive
                  ? "border-slate-900 bg-slate-900 text-white"
                  : "border-slate-200 bg-white text-slate-700"
              }`}
            >
              <div className="font-semibold text-sm">{day.dayNumber}</div>
              <div className="text-[11px] text-slate-400">Day {day.dayNumber}</div>
            </button>
          );
        })}
      </div>

      {/* Day content list */}
      <div className="flex-1 overflow-y-auto px-4">
        <p className="text-xs text-slate-500 mb-2">
          Day {activeDay.dayNumber} of {plan.totalDays}
        </p>

        <div className="space-y-2">
          {activeDay.items.map((item) => (
            <Link
              key={item.id}
              to={`/discipleship/plans/${plan.id}/day/${activeDay.dayNumber}/item/${item.id}`}
              className="flex items-center justify-between rounded-xl bg-white px-3 py-3 shadow-sm hover:shadow-md transition-shadow"
            >
              <div className="flex items-center space-x-2">
                <div className="w-5 h-5 border border-slate-300 rounded-full" />
                <div>
                  <p className="text-sm font-medium">
                    {item.type === "devotional"
                      ? "Devotional"
                      : item.reference ?? item.title}
                  </p>
                  {item.type === "scripture" && item.reference && (
                    <p className="text-[11px] text-slate-500">{item.title}</p>
                  )}
                </div>
              </div>
              <span className="text-slate-400 text-base">›</span>
            </Link>
          ))}
        </div>
      </div>

      {/* Start Reading button */}
      <div className="px-4 pb-4 pt-2">
        <button
          onClick={() => {
            const firstItem = activeDay.items[0];
            navigate(
              `/discipleship/plans/${plan.id}/day/${activeDay.dayNumber}/item/${firstItem.id}`
            );
          }}
          className="w-full rounded-full bg-slate-900 text-white py-3 text-sm font-semibold shadow-lg"
        >
          Start Reading
        </button>
      </div>
    </div>
  );
}

4️⃣ Reading page (Devotional or Scripture with meaning)
Create: client/src/features/discipleship/DiscipleshipReadingScreen.tsx
// client/src/features/discipleship/DiscipleshipReadingScreen.tsx
import { useMemo } from "react";
import { useNavigate, useParams } from "react-router-dom";
import {
  DISCIPLESHIP_PLANS,
  DiscipleshipPlan,
  PlanDay,
  PlanItem,
} from "./discipleshipPlans";

export default function DiscipleshipReadingScreen() {
  const { planId, dayNumber, itemId } = useParams<{
    planId: string;
    dayNumber: string;
    itemId: string;
  }>();
  const navigate = useNavigate();

  const plan: DiscipleshipPlan | undefined = useMemo(
    () => DISCIPLESHIP_PLANS.find((p) => p.id === planId),
    [planId]
  );

  const day: PlanDay | undefined = useMemo(() => {
    if (!plan || !dayNumber) return undefined;
    const num = Number(dayNumber);
    return plan.days.find((d) => d.dayNumber === num);
  }, [plan, dayNumber]);

  const item: PlanItem | undefined = useMemo(() => {
    if (!day || !itemId) return undefined;
    return day.items.find((i) => i.id === itemId);
  }, [day, itemId]);

  if (!plan || !day || !item) {
    return (
      <div className="p-4">
        <p className="text-sm text-red-600">Reading not found.</p>
      </div>
    );
  }

  const showReference = item.type === "scripture" && item.reference;

  return (
    <div className="flex flex-col h-full bg-slate-50">
      {/* Top bar */}
      <div className="flex items-center px-4 pt-4 pb-2">
        <button
          onClick={() => navigate(-1)}
          className="mr-3 text-sm text-slate-600"
        >
          ‹ Back
        </button>
        <div className="flex flex-col">
          <span className="text-[11px] text-slate-400 uppercase tracking-wide">
            {plan.title}
          </span>
          <h1 className="text-base font-semibold">
            Day {day.dayNumber} · {item.type === "devotional" ? "Devotional" : item.reference}
          </h1>
        </div>
      </div>

      {/* Content */}
      <div className="flex-1 overflow-y-auto px-4 pb-6">
        {showReference && (
          <p className="text-sm font-semibold text-slate-700 mt-2">
            {item.reference}
          </p>
        )}

        {/* If you want, call your Bible API here to show full verse text */}
        {/* Example (pseudo-code):
            const verseText = useBibleApi(item.reference)
        */}

        <p className="mt-3 text-[15px] leading-relaxed text-slate-800 whitespace-pre-line">
          {item.body}
        </p>

        {item.type === "devotional" && (
          <div className="mt-5 p-3 rounded-2xl bg-slate-100 text-xs text-slate-600">
            Take a moment to talk with the Lord about what you just read. Ask
            Him to help you believe and obey His Word today.
          </div>
        )}
      </div>
    </div>
  );
}

6️⃣ About “hooked to an API” / reliability
For this feature, the safest and most reliable approach is:
✅ Your plans & devotionals stored as your own data
Either bundled in the app (as above)
Or served from your own Node/Express API + database
✅ Bible text fetched from your main Bible API with Bolls as backup
✅ Everything still works even if a third-party Bible API is down:
Your explanations still show
You can show verse reference & a fallback message if text fails to load
That way:
You fully own the discipleship content
You can expand to 50+ plans easily
The user experience is stable and in your control
