import { Capacitor } from '@capacitor/core';
import { Contacts } from '@capacitor-community/contacts';

// helper tolerant of different plugin versions
function isGranted(perm: any): boolean {
  // v7 returns { granted: boolean }; older returns { contacts: 'granted'|'denied'|'prompt'|'limited' }
  if (typeof perm?.granted === 'boolean') return perm.granted;
  if (typeof perm?.contacts === 'string') return perm.contacts === 'granted' || perm.contacts === 'limited';
  return false;
}

async function openSettingsIfDenied() {
  try {
    // Deep-link to Settings if user denied
    await (window as any).Capacitor?.App?.openUrl?.({ url: 'app-settings:' });
  } catch {}
}

async function importDeviceContacts(userId: string) {
  console.log('[Contacts] Platform:', Capacitor.getPlatform(), 'userId:', userId);

  const perm = await Contacts.requestPermissions();
  console.log('[Contacts] Permission raw:', JSON.stringify(perm));
  const allowed = isGranted(perm);
  console.log('[Contacts] allowed:', allowed);

  if (!allowed) {
    console.warn('[Contacts] Permission not granted — prompting settings.');
    alert('Please enable Contacts in Settings > Privacy to import friends.');
    await openSettingsIfDenied();
    throw new Error('Contacts permission not granted');
  }

  const result = await Contacts.getContacts({
    projection: { name: true, phones: true, emails: true }
  });
  console.log('[Contacts] contacts length:', result?.contacts?.length ?? 0);

  const contacts = (result?.contacts || [])
    .map(c => ({
      firstName: c.name?.given ?? '',
      lastName: c.name?.family ?? '',
      displayName: c.displayName ?? '',
      email: (c.emails?.[0]?.address ?? '').trim() || undefined,
      phone: (c.phones?.[0]?.number ?? '').trim() || undefined,
    }))
    .filter(c => (c.firstName || c.lastName || c.displayName) && (c.email || c.phone))
    .slice(0, 50);

  console.log('[Contacts] filtered count (<=50):', contacts.length);

  if (!userId) {
    console.error('[Contacts] Missing appUserId during import!');
    alert('Please complete sign-in/registration before importing contacts.');
    throw new Error('Missing userId');
  }

  const resp = await fetch(apiUrl(`/api/contacts/${userId}/import?fromSignup=true`), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ contacts })
  });

  console.log('[Contacts] import status:', resp.status);
  const json = await resp.json().catch(() => ({}));
  console.log('[Contacts] import response:', json);

  if (!resp.ok || !json?.success) {
    throw new Error(`Import failed: ${resp.status} ${JSON.stringify(json)}`);
  }

  return json;
}

Step 2 — Harden permission handling for iOS 17/18

You already request permission, but iOS 18 can return limited. The helper above treats limited as allowed (you’ll still get a subset). If you’d rather demand full access, change:

if (typeof perm?.contacts === 'string') return perm.contacts === 'granted' || perm.contacts === 'limited';


to:

if (typeof perm?.contacts === 'string') return perm.contacts === 'granted';


…and show a UI to guide the user to Settings to switch from Limited → Full Access.

Step 4 — One common gotcha: missing appUserId

If the user hasn’t finished registration (or if you lose auth state after TestFlight install), appUserId can be empty. Add a guard where you call the import:

if (!appUserId) {
  presentToast('Please sign in first, then Import Contacts.');
  return;
}
await importDeviceContacts(appUserId);


Also ensure you persist user state (e.g., in secure storage) and rehydrate it on app launch.
