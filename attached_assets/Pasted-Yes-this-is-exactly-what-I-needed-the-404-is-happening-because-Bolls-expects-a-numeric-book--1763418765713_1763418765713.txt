
```ts
// ✅ Bolls expects numeric book ID, e.g. Romans = 45
https://bolls.life/get-verse/KJV/45/12/2/
```

So we need to:

1. Parse `"Romans 12:2"` → `"Romans"`, `12`, `2`
2. Map `"Romans"` → `45` (Bolls book ID)
3. Call `https://bolls.life/get-verse/KJV/45/12/2/`
4. Normalize Bolls response into your existing shape.

```ts
function normalizeBookName(name: string): string {
  return name.replace(/\s+/g, " ").trim();
}

function parseReference(ref: string): { book: string; chapter: number; verse: number } {
  // Example ref: "Romans 12:2", "1 Corinthians 13:4"
  const parts = ref.trim().split(" ");
  if (parts.length < 2) {
    throw new Error(`Invalid reference format: ${ref}`);
  }

  const last = parts[parts.length - 1]; // "12:2"
  const bookName = parts.slice(0, -1).join(" "); // "Romans" or "1 Corinthians"

  const [chapterStr, verseStr] = last.split(":");
  const chapter = Number(chapterStr);
  const verse = Number(verseStr);

  if (!chapter || !verse) {
    throw new Error(`Invalid chapter/verse in reference: ${ref}`);
  }

  return { book: normalizeBookName(bookName), chapter, verse };
}

function htmlToPlainText(html: string): string {
  // Very light cleanup: strip tags and convert <br> to newlines
  return html
    .replace(/<br\s*\/?>/gi, "\n")
    .replace(/<\/?[^>]+(>|$)/g, "")
    .trim();
}

// 🔹 THIS is your backup call to Bolls.life
export async function fetchFromBolls(reference: string, version: string): Promise<DailyVerseResponse> {
  const { book, chapter, verse } = parseReference(reference);

  const bookId = BOOK_NAME_TO_ID[book];
  if (!bookId) {
    throw new Error(`Unknown book name for Bolls: "${book}" from reference "${reference}"`);
  }

  const translationCode = (version || "KJV").toUpperCase();
  const translationName = TRANSLATION_NAME[translationCode] ?? translationCode;

  const url = `https://bolls.life/get-verse/${translationCode}/${bookId}/${chapter}/${verse}/`;

  const res = await fetch(url);
  if (!res.ok) {
    throw new Error(`Bolls API returned ${res.status} for ${url}`);
  }

  const json = await res.json();

  // Bolls may return a single object or an array; normalize:
  const verseData = Array.isArray(json) ? json[0] : json;

  const plainText = htmlToPlainText(verseData.text ?? "");

  return {
    success: true,
    verse: {
      text: plainText,
      reference, // e.g. "Romans 12:2"
      book,      // "Romans"
      chapter: String(chapter),
      verse: String(verse),
      translation: translationCode,
      translationName,
      date: new Date().toDateString(),
      source: "bolls",
    },
  };
}
```

```ts
import { fetchFromBolls } from "./services/bibleApiFallback";

app.get("/api/daily-verse", async (req, res) => {
  const version = (req.query.version as string) || "KJV";

  try {
    // 1️⃣ Try primary Bible API first (whatever you already have)
    const primary = await fetchFromPrimaryBibleApi(version);
    if (primary?.success) {
      return res.json(primary);
    }
  } catch (e) {
    console.warn("Primary Bible API failed, falling back to Bolls:", e);
  }

  try {
    // 2️⃣ Fallback to Bolls based on the reference you want
    // You can either:
    //  - use a predetermined daily reference, OR
    //  - use your own logic to choose a reference for today.
    const fallback = await fetchFromBolls("Romans 12:2", version);
    return res.json(fallback);
  } catch (e) {
    console.error("Bolls fallback failed:", e);
    return res.status(500).json({ success: false, error: "Unable to fetch daily verse" });
  }
});
```
