3. Code for a Verse-of-the-Day tile that looks like Bible App

Here’s a single, polished card that will look much closer to that screenshot.

File: client/src/components/VerseOfDayCard.tsx

// client/src/components/VerseOfDayCard.tsx
import { useMemo } from "react";
import { Bookmark, Share2, Heart } from "lucide-react";

interface VerseOfDayCardProps {
  reference: string;          // e.g. "John 3:16"
  text: string;               // verse text
  stats?: {
    likes?: number;
    comments?: number;
    shares?: number;
  };
  onPress?: () => void;
}

const HERO_IMAGES = [
  "/images/verse-hero-1.jpg",
  "/images/verse-hero-2.jpg",
  "/images/verse-hero-3.jpg",
];

export function VerseOfDayCard({
  reference,
  text,
  stats,
  onPress,
}: VerseOfDayCardProps) {
  const heroImage = useMemo(
    () => HERO_IMAGES[Math.floor(Math.random() * HERO_IMAGES.length)],
    []
  );

  return (
    <button
      type="button"
      onClick={onPress}
      className="w-full text-left rounded-3xl overflow-hidden bg-black/90 shadow-[0_18px_40px_rgba(0,0,0,0.6)] focus:outline-none focus:ring-2 focus:ring-amber-400/80"
    >
      {/* Image + overlay */}
      <div className="relative w-full aspect-[16/9]">
        <img
          src={heroImage}
          alt={reference}
          className="w-full h-full object-cover"
        />
        <div className="absolute inset-0 bg-gradient-to-t from-black/75 via-black/35 to-black/10" />

        <div className="absolute inset-0 flex flex-col justify-end px-4 pb-4 gap-1">
          <p className="text-xs font-semibold text-slate-200/90 mb-1">
            Verse of the Day
          </p>
          <p className="text-sm font-semibold text-amber-200">{reference}</p>
          <p className="mt-1 text-[13px] sm:text-sm leading-relaxed text-slate-50">
            {text}
          </p>
        </div>
      </div>

      {/* Stats row, similar feel to Bible app */}
      <div className="flex items-center justify-between px-4 py-2 bg-black">
        <div className="flex items-center gap-4 text-[11px] text-slate-300/80">
          <span className="inline-flex items-center gap-1.5">
            <Heart className="w-3.5 h-3.5" />
            {stats?.likes?.toLocaleString() ?? "—"}
          </span>
          <span className="inline-flex items-center gap-1.5">
            <Bookmark className="w-3.5 h-3.5" />
            {stats?.comments?.toLocaleString() ?? "—"}
          </span>
          <span className="inline-flex items-center gap-1.5">
            <Share2 className="w-3.5 h-3.5" />
            {stats?.shares?.toLocaleString() ?? "—"}
          </span>
        </div>
        <span className="text-[11px] text-slate-400">More ▾</span>
      </div>
    </button>
  );
}
