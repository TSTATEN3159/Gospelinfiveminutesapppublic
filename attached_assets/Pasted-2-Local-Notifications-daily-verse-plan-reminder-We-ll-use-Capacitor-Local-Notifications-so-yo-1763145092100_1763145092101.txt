2️⃣ Local Notifications (daily verse / plan reminder)

We’ll use Capacitor Local Notifications so your app can:

Ask permission once

Schedule daily reminders (e.g., 7:00 AM)

Let the user choose time in Settings

Step 2.1 – Install the plugin

In Replit shell (project root):

npm install @capacitor/local-notifications
npx cap sync


After npx cap sync, remember to open Xcode and let it update the project if needed.

Step 2.2 – iOS permission settings (Info.plist)

In Xcode:

Open your iOS project (usually ios/App/App.xcodeproj)

In the Info tab, add a new key:

Key: NSUserNotificationUsageDescription

Value: We use notifications to remind you of the daily verse and your Bible reading plans.

That’s the text users see when iOS asks for notification permission.

Step 2.3 – Create a notification helper (front-end)

File: src/utils/notifications.ts

import {
  LocalNotifications,
  PermissionStatus,
} from "@capacitor/local-notifications";

export async function requestNotificationPermission(): Promise<boolean> {
  // Check existing permission
  const permStatus: PermissionStatus = await LocalNotifications.checkPermissions();

  if (permStatus.display === "granted") {
    return true;
  }

  // Ask if not yet granted
  const newStatus = await LocalNotifications.requestPermissions();
  return newStatus.display === "granted";
}

type ScheduleDailyReminderOptions = {
  id: number;
  hour: number;   // 0–23
  minute: number; // 0–59
  title: string;
  body: string;
};

export async function scheduleDailyReminder(options: ScheduleDailyReminderOptions) {
  const granted = await requestNotificationPermission();
  if (!granted) {
    throw new Error("Notification permission not granted");
  }

  const { id, hour, minute, title, body } = options;

  await LocalNotifications.schedule({
    notifications: [
      {
        id,
        title,
        body,
        schedule: {
          repeats: true,
          on: { hour, minute },
        },
        sound: undefined, // use default
      },
    ],
  });
}

export async function cancelReminder(id: number) {
  await LocalNotifications.cancel({ notifications: [{ id }] });
}


We’ll use:

id = 1 → Verse of the Day reminder

id = 2 → Plan reminder

You can expand later.

Step 2.4 – Settings UI to control reminders

Let’s add a simple “Reminders” section to your Settings page under the “More” tab.

File: src/components/ReminderSettings.tsx

import React, { useEffect, useState } from "react";
import {
  scheduleDailyReminder,
  cancelReminder,
} from "../utils/notifications";

type Time = { hour: number; minute: number };

const STORAGE_KEY_VERSE_REMINDER = "verse-reminder-time";

function parseStoredTime(): Time | null {
  if (typeof window === "undefined") return null;
  const stored = window.localStorage.getItem(STORAGE_KEY_VERSE_REMINDER);
  if (!stored) return null;
  try {
    const parsed = JSON.parse(stored) as Time;
    if (
      typeof parsed.hour === "number" &&
      typeof parsed.minute === "number"
    ) {
      return parsed;
    }
  } catch {
    // ignore
  }
  return null;
}

export const ReminderSettings: React.FC = () => {
  const [enabled, setEnabled] = useState(false);
  const [time, setTime] = useState<Time>({ hour: 7, minute: 0 });
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (typeof window === "undefined") return;

    const storedTime = parseStoredTime();
    if (storedTime) {
      setTime(storedTime);
      setEnabled(true);
    }
  }, []);

  const handleTimeChange = (value: string) => {
    const [h, m] = value.split(":");
    const hour = Number(h);
    const minute = Number(m);
    if (!Number.isNaN(hour) && !Number.isNaN(minute)) {
      setTime({ hour, minute });
    }
  };

  const timeString = `${String(time.hour).padStart(2, "0")}:${String(
    time.minute
  ).padStart(2, "0")}`;

  const handleSave = async () => {
    setSaving(true);
    setError(null);
    try {
      if (enabled) {
        await scheduleDailyReminder({
          id: 1,
          hour: time.hour,
          minute: time.minute,
          title: "Daily Verse",
          body: "Your verse of the day is ready. Tap to read and meditate.",
        });

        if (typeof window !== "undefined") {
          window.localStorage.setItem(
            STORAGE_KEY_VERSE_REMINDER,
            JSON.stringify(time)
          );
        }
      } else {
        await cancelReminder(1);
        if (typeof window !== "undefined") {
          window.localStorage.removeItem(STORAGE_KEY_VERSE_REMINDER);
        }
      }
    } catch (err: any) {
      console.error("Error saving reminder", err);
      setError(
        err?.message ||
          "Sorry, something went wrong updating your reminder settings."
      );
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="space-y-3">
      <h2 className="text-lg font-semibold">Daily Verse Reminder</h2>
      {error && (
        <div className="bg-red-600 text-white rounded-md p-2 text-sm">
          {error}
        </div>
      )}
      <div className="flex items-center justify-between">
        <span>Enable Reminder</span>
        <button
          type="button"
          onClick={() => setEnabled((prev) => !prev)}
          className={`w-12 h-6 rounded-full p-1 transition ${
            enabled ? "bg-green-500" : "bg-gray-300"
          }`}
        >
          <div
            className={`w-4 h-4 bg-white rounded-full transform transition ${
              enabled ? "translate-x-6" : "translate-x-0"
            }`}
          />
        </button>
      </div>

      <div className="flex items-center justify-between">
        <label htmlFor="reminder-time">Reminder Time</label>
        <input
          id="reminder-time"
          type="time"
          value={timeString}
          onChange={(e) => handleTimeChange(e.target.value)}
          disabled={!enabled}
          className="border rounded-md px-2 py-1"
        />
      </div>

      <button
        type="button"
        onClick={handleSave}
        disabled={saving}
        className="w-full mt-2 bg-black text-white rounded-lg py-2 font-semibold"
      >
        {saving ? "Saving..." : "Save Reminder"}
      </button>
    </div>
  );
};
