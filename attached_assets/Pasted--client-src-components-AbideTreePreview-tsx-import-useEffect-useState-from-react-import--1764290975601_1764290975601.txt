// client/src/components/AbideTreePreview.tsx
import { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";
import { Leaf, Flame } from "lucide-react";
import { tapHaptic } from "@/lib/nativeEnhancements";
import { getAbideState } from "@/lib/abideStorage";

export default function AbideTreePreview() {
  const navigate = useNavigate();
  const [state, setState] = useState(getAbideState());
  const [justGrew, setJustGrew] = useState(false);

  useEffect(() => {
    const s = getAbideState();
    setState(s);

    if (s.totalFruit > 0) {
      setJustGrew(true);
      const t = setTimeout(() => setJustGrew(false), 450);
      return () => clearTimeout(t);
    }
  }, []);

  const handleTap = async () => {
    await tapHaptic();
    navigate("/abide");
  };

  const fruits = state.fruitCounts;
  const maxSlots = 12;

  return (
    <button
      onClick={handleTap}
      className="
        relative w-full 
        overflow-hidden rounded-3xl
        bg-gradient-to-b from-emerald-900 via-emerald-950 to-black
        border border-emerald-900/70
        shadow-[0_25px_65px_rgba(0,0,0,0.65)]
        backdrop-blur-xl
        px-6 py-6 mb-7
        text-left 
        transition-transform active:scale-[0.98]
      "
    >
      {/* Background Light & Glow */}
      <div className="absolute inset-0">
        <div className="absolute top-0 inset-x-0 h-32 bg-emerald-500/15 blur-2xl" />
      </div>

      {/* Top Label */}
      <p className="text-xs uppercase tracking-[0.18em] text-emerald-200/70 mb-2">
        ABIDE
      </p>

      {/* Main Title */}
      <h2 className="text-[18px] font-semibold text-emerald-50 leading-tight mb-1">
        My Growth Today
      </h2>

      {/* Fruit type + streak */}
      <div className="flex justify-between items-center mb-4">
        <span className="flex items-center gap-1 text-[14px] text-emerald-200">
          <Leaf className="h-4 w-4 text-emerald-300" />
          {state.lastAbideDate ? (
            <>Fruit: <span className="font-medium">{getTodayFruitLabel(fruits)}</span></>
          ) : (
            <>Begin your first Fruit</>
          )}
        </span>

        <span className="flex items-center gap-1 text-[12px] text-amber-300">
          <Flame className="h-3 w-3" />
          {state.streak > 0 ? `${state.streak} Day Streak` : "Start now"}
        </span>
      </div>

      {/* Fruit Display - Larger, More Premium */}
      <div className="flex items-center mt-3 gap-2 flex-wrap">
        {Array.from({ length: Math.min(state.totalFruit, maxSlots) }).map((_, i) => (
          <div
            key={i}
            className={`
              h-6 w-6 rounded-full bg-amber-400 shadow-lg shadow-amber-900/40
              border border-amber-600/60
              ${justGrew && i === state.totalFruit - 1 ? "fruit-pop" : ""}
            `}
          />
        ))}

        {Array.from({
          length: Math.max(0, maxSlots - Math.min(state.totalFruit, maxSlots)),
        }).map((_, i) => (
          <div
            key={`e-${i}`}
            className="h-6 w-6 rounded-full border border-amber-600/20 opacity-30"
          />
        ))}
      </div>

      {/* CTA */}
      <p className="mt-5 text-[12px] font-medium text-emerald-200/70">
        Tap to Abide — 5 minutes with Jesus
      </p>
    </button>
  );
}

function getTodayFruitLabel(fruits: Record<string, number>) {
  const keys = Object.keys(fruits);
  const sum = keys.reduce((acc, k) => acc + fruits[k], 0);
  return keys[sum % keys.length] || "Love";
}
