1️⃣ Bookmark storage service (folders + notes + bookmarks)

File: client/src/services/bookmarkService.ts

// client/src/services/bookmarkService.ts

const STORAGE_KEY = "bibleBookmarks_v2";

export interface BookmarkFolder {
  id: string;
  name: string;
  createdAt: string;
}

export interface BookmarkItem {
  id: string;
  reference: string;      // e.g. "John 3:16-18"
  folderId: string;
  note: string;
  createdAt: string;
  updatedAt: string;
}

interface BookmarkState {
  folders: BookmarkFolder[];
  bookmarks: BookmarkItem[];
}

function loadState(): BookmarkState {
  if (typeof window === "undefined") {
    return { folders: [], bookmarks: [] };
  }

  const raw = window.localStorage.getItem(STORAGE_KEY);
  if (!raw) {
    // create a default folder on first run
    const defaultFolder: BookmarkFolder = {
      id: "default",
      name: "My Verses",
      createdAt: new Date().toISOString(),
    };
    const initial: BookmarkState = { folders: [defaultFolder], bookmarks: [] };
    saveState(initial);
    return initial;
  }

  try {
    const parsed = JSON.parse(raw) as BookmarkState;
    return {
      folders: parsed.folders ?? [],
      bookmarks: parsed.bookmarks ?? [],
    };
  } catch {
    const fallback: BookmarkState = { folders: [], bookmarks: [] };
    saveState(fallback);
    return fallback;
  }
}

function saveState(state: BookmarkState) {
  if (typeof window === "undefined") return;
  window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
}

export function getFolders(): BookmarkFolder[] {
  return loadState().folders;
}

export function getBookmarksForFolder(folderId: string): BookmarkItem[] {
  const state = loadState();
  return state.bookmarks
    .filter((b) => b.folderId === folderId)
    .sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
}

export function addFolder(name: string): BookmarkFolder {
  const state = loadState();
  const id = `folder-${Date.now()}`;

  const folder: BookmarkFolder = {
    id,
    name: name.trim() || "New Folder",
    createdAt: new Date().toISOString(),
  };

  state.folders.push(folder);
  saveState(state);
  return folder;
}

export function renameFolder(id: string, newName: string) {
  const state = loadState();
  const folder = state.folders.find((f) => f.id === id);
  if (!folder) return;

  folder.name = newName.trim() || folder.name;
  saveState(state);
}

export function deleteFolder(id: string) {
  const state = loadState();
  state.folders = state.folders.filter((f) => f.id !== id);
  state.bookmarks = state.bookmarks.filter((b) => b.folderId !== id);
  saveState(state);
}

export function isBookmarked(reference: string, folderId?: string): boolean {
  const state = loadState();
  if (folderId) {
    return state.bookmarks.some(
      (b) => b.reference === reference && b.folderId === folderId
    );
  }
  // any folder
  return state.bookmarks.some((b) => b.reference === reference);
}

// Add or toggle bookmark in a given folder.
// If a bookmark with same reference+folder exists → remove it.
// If not → create it with optional note.
export function toggleBookmark(
  reference: string,
  folderId: string,
  initialNote: string = ""
): { added: boolean; item?: BookmarkItem } {
  const state = loadState();

  const existing = state.bookmarks.find(
    (b) => b.reference === reference && b.folderId === folderId
  );

  if (existing) {
    state.bookmarks = state.bookmarks.filter((b) => b.id !== existing.id);
    saveState(state);
    return { added: false };
  }

  const item: BookmarkItem = {
    id: `bm-${Date.now()}`,
    reference,
    folderId,
    note: initialNote,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  };

  state.bookmarks.push(item);
  saveState(state);
  return { added: true, item };
}

export function updateBookmarkNote(id: string, note: string) {
  const state = loadState();
  const item = state.bookmarks.find((b) => b.id === id);
  if (!item) return;
  item.note = note;
  item.updatedAt = new Date().toISOString();
  saveState(state);
}

export function deleteBookmark(id: string) {
  const state = loadState();
  state.bookmarks = state.bookmarks.filter((b) => b.id !== id);
  saveState(state);
}

2️⃣ Bookmarks page with folders + notes

This is a full page under your “More” tab.

File: client/src/pages/BookmarksPage.tsx

import { useEffect, useState } from "react";
import {
  getFolders,
  getBookmarksForFolder,
  addFolder,
  renameFolder,
  deleteFolder,
  updateBookmarkNote,
  deleteBookmark,
  BookmarkFolder,
  BookmarkItem,
} from "@/services/bookmarkService";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";
import { FolderPlus, Trash2, Edit2, BookOpen } from "lucide-react";

interface BookmarksPageProps {
  onNavigateToReference?: (reference: string) => void;
}

export default function BookmarksPage({ onNavigateToReference }: BookmarksPageProps) {
  const [folders, setFolders] = useState<BookmarkFolder[]>([]);
  const [activeFolderId, setActiveFolderId] = useState<string | null>(null);
  const [bookmarks, setBookmarks] = useState<BookmarkItem[]>([]);
  const [newFolderName, setNewFolderName] = useState("");
  const [editingFolderId, setEditingFolderId] = useState<string | null>(null);
  const [editingFolderName, setEditingFolderName] = useState("");
  const [editingNoteId, setEditingNoteId] = useState<string | null>(null);
  const [editingNoteText, setEditingNoteText] = useState("");

  const refresh = () => {
    const f = getFolders();
    setFolders(f);
    if (!activeFolderId && f.length > 0) {
      setActiveFolderId(f[0].id);
      setBookmarks(getBookmarksForFolder(f[0].id));
    } else if (activeFolderId) {
      setBookmarks(getBookmarksForFolder(activeFolderId));
    }
  };

  useEffect(() => {
    refresh();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  useEffect(() => {
    if (activeFolderId) {
      setBookmarks(getBookmarksForFolder(activeFolderId));
    } else {
      setBookmarks([]);
    }
  }, [activeFolderId]);

  const handleAddFolder = () => {
    if (!newFolderName.trim()) return;
    const folder = addFolder(newFolderName.trim());
    setNewFolderName("");
    setFolders(getFolders());
    setActiveFolderId(folder.id);
  };

  const startEditFolder = (folder: BookmarkFolder) => {
    setEditingFolderId(folder.id);
    setEditingFolderName(folder.name);
  };

  const saveFolderName = () => {
    if (!editingFolderId) return;
    renameFolder(editingFolderId, editingFolderName);
    setEditingFolderId(null);
    setEditingFolderName("");
    setFolders(getFolders());
  };

  const handleDeleteFolder = (id: string) => {
    deleteFolder(id);
    const f = getFolders();
    setFolders(f);
    if (activeFolderId === id) {
      setActiveFolderId(f.length ? f[0].id : null);
    }
  };

  const startEditNote = (bookmark: BookmarkItem) => {
    setEditingNoteId(bookmark.id);
    setEditingNoteText(bookmark.note || "");
  };

  const saveNote = () => {
    if (!editingNoteId) return;
    updateBookmarkNote(editingNoteId, editingNoteText);
    setEditingNoteId(null);
    setEditingNoteText("");
    if (activeFolderId) {
      setBookmarks(getBookmarksForFolder(activeFolderId));
    }
  };

  const handleDeleteBookmark = (id: string) => {
    deleteBookmark(id);
    if (activeFolderId) {
      setBookmarks(getBookmarksForFolder(activeFolderId));
    }
  };

  return (
    <div className="min-h-screen bg-gradient-to-b from-slate-50 to-white dark:from-slate-900 dark:to-slate-950 pb-20">
      <div className="px-4 pt-6 pb-4 border-b border-slate-200 dark:border-slate-800 bg-white/80 dark:bg-slate-900/80 backdrop-blur">
        <h1 className="text-2xl font-bold text-slate-900 dark:text-slate-50">
          Bookmarks & Notes
        </h1>
        <p className="text-sm text-slate-500 dark:text-slate-400">
          Save Scriptures in folders and add personal notes for meditation and teaching.
        </p>
      </div>

      <div className="p-4 max-w-4xl mx-auto grid md:grid-cols-[240px,1fr] gap-4">
        {/* Folders column */}
        <Card className="border-slate-200 dark:border-slate-800">
          <CardHeader className="pb-2">
            <CardTitle className="text-base">Folders</CardTitle>
          </CardHeader>
          <CardContent className="space-y-3">
            <div className="flex gap-2">
              <Input
                placeholder="New folder name"
                value={newFolderName}
                onChange={(e) => setNewFolderName(e.target.value)}
                className="text-sm"
              />
              <Button size="icon" onClick={handleAddFolder}>
                <FolderPlus className="w-4 h-4" />
              </Button>
            </div>

            <div className="space-y-1 max-h-[320px] overflow-y-auto">
              {folders.map((folder) => {
                const isActive = folder.id === activeFolderId;
                const bookmarkCount = getBookmarksForFolder(folder.id).length;

                return (
                  <div
                    key={folder.id}
                    className={`flex items-center justify-between rounded-lg px-3 py-2 text-sm cursor-pointer ${
                      isActive
                        ? "bg-blue-600 text-white"
                        : "bg-slate-100/70 dark:bg-slate-800/70 text-slate-800 dark:text-slate-100"
                    }`}
                    onClick={() => setActiveFolderId(folder.id)}
                  >
                    <div className="flex flex-col">
                      {editingFolderId === folder.id ? (
                        <Input
                          value={editingFolderName}
                          onChange={(e) => setEditingFolderName(e.target.value)}
                          className={`h-7 text-xs ${
                            isActive
                              ? "bg-white/10 text-white"
                              : "bg-white text-slate-900"
                          }`}
                          onBlur={saveFolderName}
                          onKeyDown={(e) => e.key === "Enter" && saveFolderName()}
                          autoFocus
                        />
                      ) : (
                        <>
                          <span className="font-semibold truncate">
                            {folder.name}
                          </span>
                          <span
                            className={`text-[10px] ${
                              isActive
                                ? "text-blue-100"
                                : "text-slate-500 dark:text-slate-400"
                            }`}
                          >
                            {bookmarkCount} verse{bookmarkCount === 1 ? "" : "s"}
                          </span>
                        </>
                      )}
                    </div>

                    <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
                      <button
                        className="p-1 rounded-full hover:bg-black/10"
                        onClick={() => startEditFolder(folder)}
                      >
                        <Edit2 className="w-3 h-3" />
                      </button>
                      <button
                        className="p-1 rounded-full hover:bg-black/10"
                        onClick={() => handleDeleteFolder(folder.id)}
                      >
                        <Trash2 className="w-3 h-3" />
                      </button>
                    </div>
                  </div>
                );
              })}

              {folders.length === 0 && (
                <p className="text-xs text-slate-500">
                  Create a folder to start organizing your favorite Scriptures.
                </p>
              )}
            </div>
          </CardContent>
        </Card>

        {/* Bookmarks / notes column */}
        <Card className="border-slate-200 dark:border-slate-800">
          <CardHeader className="pb-2">
            <CardTitle className="text-base">
              {activeFolderId
                ? `Bookmarks in "${
                    folders.find((f) => f.id === activeFolderId)?.name ?? ""
                  }"`
                : "Bookmarks"}
            </CardTitle>
          </CardHeader>
          <CardContent className="space-y-3 max-h-[640px] overflow-y-auto">
            {bookmarks.length === 0 && (
              <p className="text-sm text-slate-500">
                No verses saved yet in this folder. In Scripture view, tap the ⭐ bookmark button to add one.
              </p>
            )}

            {bookmarks.map((bm) => (
              <div
                key={bm.id}
                className="rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 p-3"
              >
                <div className="flex items-center justify-between mb-2">
                  <div className="flex items-center gap-2">
                    <button
                      className="inline-flex items-center gap-1 text-xs font-semibold text-blue-600 dark:text-blue-300 hover:underline"
                      onClick={() =>
                        onNavigateToReference && onNavigateToReference(bm.reference)
                      }
                    >
                      <BookOpen className="w-3 h-3" />
                      {bm.reference}
                    </button>
                  </div>
                  <button
                    className="p-1 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800"
                    onClick={() => handleDeleteBookmark(bm.id)}
                  >
                    <Trash2 className="w-4 h-4 text-slate-500" />
                  </button>
                </div>

                {editingNoteId === bm.id ? (
                  <div className="space-y-2">
                    <Textarea
                      autoFocus
                      rows={3}
                      value={editingNoteText}
                      onChange={(e) => setEditingNoteText(e.target.value)}
                      className="text-sm"
                      placeholder="Write what the Lord showed you in this passage..."
                    />
                    <div className="flex justify-end gap-2">
                      <Button
                        size="sm"
                        variant="outline"
                        onClick={() => {
                          setEditingNoteId(null);
                          setEditingNoteText("");
                        }}
                      >
                        Cancel
                      </Button>
                      <Button size="sm" onClick={saveNote}>
                        Save Note
                      </Button>
                    </div>
                  </div>
                ) : (
                  <div
                    className="text-sm text-slate-700 dark:text-slate-200 whitespace-pre-wrap"
                    onClick={() => startEditNote(bm)}
                  >
                    {bm.note
                      ? bm.note
                      : <span className="italic text-slate-400">Tap here to add a personal note…</span>}
                  </div>
                )}

                <div className="mt-2 text-[10px] text-slate-400">
                  Saved: {new Date(bm.createdAt).toLocaleString()}
                  {bm.updatedAt !== bm.createdAt && (
                    <> • Updated: {new Date(bm.updatedAt).toLocaleString()}</>
                  )}
                </div>
              </div>
            ))}
          </CardContent>
        </Card>
      </div>
    </div>
  );
}

3️⃣ Hook the ⭐ bookmark button in your Scripture view

Wherever you render a reference (e.g. "John 3:16-18"), add a bookmark button that uses the service.

Example inside your Scripture display component:

import { toggleBookmark } from "@/services/bookmarkService";
import { Star } from "lucide-react";
import { useState } from "react";

interface BookmarkStarProps {
  reference: string;
  folderId?: string; // optional; default "default"
}

export function BookmarkStar({ reference, folderId = "default" }: BookmarkStarProps) {
  const [isOn, setIsOn] = useState(false);

  const handleClick = () => {
    const result = toggleBookmark(reference, folderId);
    setIsOn(result.added);
  };

  return (
    <button
      onClick={handleClick}
      className="ml-2 inline-flex items-center justify-center p-1 rounded-full hover:bg-yellow-50"
      title={isOn ? "Remove bookmark" : "Save verse"}
    >
      <Star
        className="w-4 h-4"
        style={{ color: isOn ? "#eab308" : "#9ca3af" }} // yellow-500 / gray-400
        fill={isOn ? "#eab308" : "none"}
      />
    </button>
  );
}


Then near the reference label:

<div className="flex items-center">
  <span className="text-xs font-semibold text-slate-500">
    {reference}
  </span>
  <BookmarkStar reference={reference} />
</div>
