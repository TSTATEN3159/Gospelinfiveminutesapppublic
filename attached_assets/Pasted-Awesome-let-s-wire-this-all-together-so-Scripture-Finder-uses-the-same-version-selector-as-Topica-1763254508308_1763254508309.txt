1️⃣ Shared Bible Version Config (one source of truth)

File (new): client/src/config/bibleVersions.ts

// client/src/config/bibleVersions.ts

export type BibleVersionCode =
  | "KJV"
  | "BBE"
  | "WEB"
  | "ASV"; // extend with more later

export interface BibleVersionDef {
  code: BibleVersionCode;
  label: string;
}

export const BIBLE_VERSIONS: BibleVersionDef[] = [
  { code: "KJV", label: "KJV – King James Version" },
  { code: "BBE", label: "BBE – Bible in Basic English" },
  { code: "WEB", label: "WEB – World English Bible" },
  { code: "ASV", label: "ASV – American Standard Version" }
];

const STORAGE_KEY_VERSION = "preferredBibleVersion";

export function getInitialBibleVersion(): BibleVersionCode {
  if (typeof window === "undefined") return "KJV";
  const stored = window.localStorage.getItem(STORAGE_KEY_VERSION) as BibleVersionCode | null;
  if (stored && BIBLE_VERSIONS.some(v => v.code === stored)) {
    return stored;
  }
  return "KJV";
}

export function setPreferredBibleVersion(version: BibleVersionCode) {
  if (typeof window === "undefined") return;
  window.localStorage.setItem(STORAGE_KEY_VERSION, version);
}

2️⃣ Reusable Version Selector UI

File (new): client/src/components/BibleVersionSelector.tsx

// client/src/components/BibleVersionSelector.tsx
import {
  BIBLE_VERSIONS,
  BibleVersionCode,
  getInitialBibleVersion,
  setPreferredBibleVersion
} from "@/config/bibleVersions";
import {
  Select,
  SelectTrigger,
  SelectContent,
  SelectItem,
  SelectValue
} from "@/components/ui/select";
import { useEffect, useState } from "react";

interface BibleVersionSelectorProps {
  label?: string;
  onChange?: (version: BibleVersionCode) => void;
  value?: BibleVersionCode;   // optional controlled
}

export function BibleVersionSelector({ label = "Bible Version", onChange, value }: BibleVersionSelectorProps) {
  const [internalValue, setInternalValue] = useState<BibleVersionCode>(getInitialBibleVersion());

  const current = value ?? internalValue;

  useEffect(() => {
    if (!value) {
      // keep internal value synced with stored
      setInternalValue(getInitialBibleVersion());
    }
  }, [value]);

  const handleChange = (v: string) => {
    const code = v as BibleVersionCode;
    if (!value) {
      setInternalValue(code);
      setPreferredBibleVersion(code);
    }
    onChange?.(code);
  };

  return (
    <div className="w-full">
      {label && (
        <div className="text-xs font-semibold text-slate-500 dark:text-slate-400 mb-1">
          {label}
        </div>
      )}
      <Select value={current} onValueChange={handleChange}>
        <SelectTrigger className="w-full bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700">
          <SelectValue placeholder="Choose version" />
        </SelectTrigger>
        <SelectContent>
          {BIBLE_VERSIONS.map(v => (
            <SelectItem key={v.code} value={v.code}>
              {v.label}
            </SelectItem>
          ))}
        </SelectContent>
      </Select>
    </div>
  );
}


Now Scripture Finder and Topical Bible Search can share this exact UI and preference.

You likely already have:

A search form (Book / Chapter / Verse / Range / etc.)

At the top of BibleSearchPage.tsx:

B. Drop the version selector into the UI

Where you render the search controls (near the Book/Chapter/Verse selectors), add:

  <BibleVersionSelector
    label="Bible Version"
    value={selectedVersion}
    onChange={(v) => setSelectedVersion(v)}
  />

Visually, this will now behave like the Topical search: same look, same behavior, and same stored preference.

C. Use selectedVersion in the API call

Wherever you fetch scripture text, include version as a query or body parameter.

Example (adjust to your actual endpoint):

async function fetchScripture(reference: string) {
  setLoading(true);
  setError(null);

  try {
    const res = await fetch(
      `/api/bible-passage?reference=${encodeURIComponent(reference)}&version=${encodeURIComponent(selectedVersion)}`
    );

    if (!res.ok) {
      throw new Error("Failed to fetch passage");
    }

    const data = await res.json(); // { reference, text, version? }
    setScriptureText(data.text ?? "");
    setScriptureReference(data.reference ?? reference);
  } catch (err: any) {
    console.error(err);
    setError("Sorry, I couldn't load that passage right now.");
  } finally {
    setLoading(false);
  }
}


Then, when the user submits the form:

const handleSubmit = (e: React.FormEvent) => {
  e.preventDefault();
  const ref = buildReferenceFromControls(); // e.g. "John 3:16-18"
  fetchScripture(ref);
};

4️⃣ “Send to Image Generator” from Scripture Finder

Assume:

Your page receives onNavigate?: (page: string, params?: any) => void

Your Scripture image creator lives on some route like "image-scripture" or "image-generator" and expects { reference, text }.

A. Add a handler in BibleSearchPage.tsx

Below your other handlers:

const handleSendToImage = () => {
  if (!scriptureText || !scriptureReference) return;

  // Navigate to your image-creation flow with the verse pre-filled
  onNavigate?.("image-scripture", {
    reference: scriptureReference,
    text: scriptureText,
    version: selectedVersion
  });
};

B. Add a button near the scripture result

Where you show the verse (after fetch):

{scriptureText && (
  <div className="mt-4 space-y-3">
    <div className="flex items-center justify-between">
      <div className="text-xs font-semibold text-slate-500">
        {scriptureReference} • {selectedVersion}
      </div>
      <Button
        size="sm"
        variant="outline"
        onClick={handleSendToImage}
      >
        Create Scripture Image
      </Button>
    </div>

    <div className="rounded-xl bg-slate-50 dark:bg-slate-900 p-4 text-sm leading-relaxed">
      {scriptureText}
    </div>
  </div>
)}


When tapped, this routes the user (with the verse data) into your Image Scripture Generator page.
