📌 STEP 2 — Add the Dropdown UI

File: client/src/pages/TopicSearchPage.tsx

import { useState } from "react";
import { BIBLE_TOPICS, TopicData } from "@/data/topics";
import { Card, CardHeader, CardTitle, CardContent } from "@/components/ui/card";
import { Select, SelectTrigger, SelectContent, SelectItem, SelectValue } from "@/components/ui/select";

export default function TopicSearchPage() {
  const [selectedTopic, setSelectedTopic] = useState<TopicData | null>(null);

  const onTopicSelect = (topicName: string) => {
    const topic = BIBLE_TOPICS.find(t => t.topic === topicName) || null;
    setSelectedTopic(topic);
  };

  return (
    <div className="min-h-screen px-4 py-6 bg-gradient-to-b from-amber-50 to-white dark:from-gray-900 dark:to-gray-800">
      
      <h1 className="text-3xl font-bold mb-4 text-center">
        Topical Bible Search
      </h1>

      {/* Dropdown */}
      <div className="mb-6 max-w-md mx-auto">
        <Select onValueChange={onTopicSelect}>
          <SelectTrigger className="w-full py-3 bg-white shadow-lg dark:bg-gray-800 border-2 border-amber-400 rounded-xl">
            <SelectValue placeholder="Choose a Biblical Topic" />
          </SelectTrigger>
          <SelectContent>
            {BIBLE_TOPICS.map((item) => (
              <SelectItem key={item.topic} value={item.topic}>
                {item.topic}
              </SelectItem>
            ))}
          </SelectContent>
        </Select>
      </div>

      {/* Results */}
      {selectedTopic && (
        <Card className="max-w-2xl mx-auto shadow-xl border-amber-300 mb-10">
          <CardHeader>
            <CardTitle className="text-xl font-semibold">
              Scriptures on {selectedTopic.topic}
            </CardTitle>
          </CardHeader>
          <CardContent className="space-y-4">
            {selectedTopic.verses.map((v, idx) => (
              <div
                key={idx}
                className="p-3 rounded-lg bg-amber-50 dark:bg-gray-800 border-l-4 border-amber-500 cursor-pointer hover:bg-amber-100"
              >
                <p className="font-semibold">{v.ref}</p>
                <p className="text-sm opacity-80">{v.text}</p>
              </div>
            ))}
          </CardContent>
        </Card>
      )}
    </div>
  );
}
