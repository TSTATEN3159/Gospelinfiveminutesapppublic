2. Topical Bible Search tile component

Create/update:

client/src/components/TopicalBibleSearchCard.tsx

// client/src/components/TopicalBibleSearchCard.tsx
import { useEffect, useMemo, useState } from "react";
import { Lightbulb, ChevronDown, BookOpen, Loader2 } from "lucide-react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue
} from "@/components/ui/select";
import { ScrollArea } from "@/components/ui/scroll-area";

import {
  TOPICAL_BIBLE_TOPICS,
  TopicDefinition,
  BibleVersionCode
} from "@/data/topicalBibleTopics";

interface VerseResult {
  reference: string;
  text: string;
  version: BibleVersionCode;
}

const BIBLE_VERSIONS: { code: BibleVersionCode; label: string }[] = [
  { code: "KJV", label: "KJV – King James Version" },
  { code: "BBE", label: "BBE – Bible in Basic English" },
  { code: "WEB", label: "WEB – World English Bible" },
  { code: "ASV", label: "ASV – American Standard Version" }
  // add more as you wire licensing + backend
];

export function TopicalBibleSearchCard() {
  const [selectedTopicId, setSelectedTopicId] = useState<string | null>(null);
  const [selectedVersion, setSelectedVersion] =
    useState<BibleVersionCode>("KJV");
  const [verses, setVerses] = useState<VerseResult[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const selectedTopic: TopicDefinition | null = useMemo(
    () =>
      TOPICAL_BIBLE_TOPICS.find((t) => t.id === selectedTopicId) ??
      TOPICAL_BIBLE_TOPICS[0] ??
      null,
    [selectedTopicId]
  );

  // Ensure we always have a topic selected
  useEffect(() => {
    if (!selectedTopicId && TOPICAL_BIBLE_TOPICS.length > 0) {
      setSelectedTopicId(TOPICAL_BIBLE_TOPICS[0].id);
    }
  }, [selectedTopicId]);

  // Fetch all verses for the selected topic + version
  const loadVerses = async (topic: TopicDefinition, version: BibleVersionCode) =>
  {
    try {
      setLoading(true);
      setError(null);

      const results: VerseResult[] = [];

      // Simple sequential calls; you can optimize to batch on the backend
      for (const reference of topic.references) {
        const res = await fetch(
          `/api/bible-verse?reference=${encodeURIComponent(
            reference
          )}&version=${encodeURIComponent(version)}`
        );

        if (!res.ok) {
          throw new Error("Failed to fetch verse text");
        }

        const data = (await res.json()) as { reference: string; text: string };

        results.push({
          reference: data.reference ?? reference,
          text: data.text ?? "",
          version
        });
      }

      setVerses(results);
    } catch (err: any) {
      console.error("Failed to load topical verses", err);
      setError(
        "Sorry, I'm having trouble loading the verses right now. Please try again."
      );
      setVerses([]);
    } finally {
      setLoading(false);
    }
  };

  // Reload when topic or version changes
  useEffect(() => {
    if (selectedTopic) {
      void loadVerses(selectedTopic, selectedVersion);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [selectedTopicId, selectedVersion]);

  if (!selectedTopic) return null;

  return (
    <Card className="shadow-xl border-0 overflow-hidden bg-gradient-to-b from-amber-50 to-rose-50 dark:from-slate-900 dark:to-slate-950">
      <CardHeader className="flex flex-row items-start gap-4 bg-gradient-to-r from-amber-400 to-rose-400 text-white">
        <div className="mt-1">
          <div className="w-12 h-12 rounded-full bg-white/15 flex items-center justify-center">
            <Lightbulb className="w-6 h-6" />
          </div>
        </div>
        <div className="flex-1 space-y-1">
          <CardTitle className="text-2xl font-semibold">
            Topical Bible Search
          </CardTitle>
          <p className="text-sm text-amber-50/90">
            Explore key Biblical topics and discover God&apos;s wisdom.
          </p>
        </div>
      </CardHeader>

      <CardContent className="p-5 space-y-5">
        {/* Controls row */}
        <div className="flex flex-col md:flex-row gap-3 md:items-center md:justify-between">
          {/* Topic dropdown */}
          <div className="flex-1">
            <div className="text-xs font-semibold text-slate-500 dark:text-slate-400 mb-1">
              Topic
            </div>
            <Select
              value={selectedTopicId ?? undefined}
              onValueChange={(value) => setSelectedTopicId(value)}
            >
              <SelectTrigger className="w-full bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700">
                <SelectValue placeholder="Choose a topic" />
              </SelectTrigger>
              <SelectContent className="max-h-72">
                {TOPICAL_BIBLE_TOPICS.map((topic) => (
                  <SelectItem key={topic.id} value={topic.id}>
                    {topic.name}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          {/* Version dropdown */}
          <div className="w-full md:w-64">
            <div className="text-xs font-semibold text-slate-500 dark:text-slate-400 mb-1">
              Bible Version
            </div>
            <Select
              value={selectedVersion}
              onValueChange={(value) =>
                setSelectedVersion(value as BibleVersionCode)
              }
            >
              <SelectTrigger className="w-full bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700">
                <SelectValue placeholder="Choose version" />
              </SelectTrigger>
              <SelectContent>
                {BIBLE_VERSIONS.map((v) => (
                  <SelectItem key={v.code} value={v.code}>
                    {v.label}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>
        </div>

        {/* Topic description */}
        <div className="rounded-2xl bg-white/70 dark:bg-slate-900/70 border border-amber-100 dark:border-slate-700 px-4 py-3 flex items-center gap-3">
          <div className="w-8 h-8 rounded-full bg-amber-100 dark:bg-amber-900/40 flex items-center justify-center">
            <BookOpen className="w-4 h-4 text-amber-700 dark:text-amber-200" />
          </div>
          <div>
            <div className="text-sm font-semibold text-slate-800 dark:text-slate-100">
              {selectedTopic.name}
            </div>
            <div className="text-xs text-slate-600 dark:text-slate-400">
              {selectedTopic.description}
            </div>
          </div>
        </div>

        {/* Error */}
        {error && (
          <div className="rounded-xl bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 px-4 py-3 text-sm text-red-800 dark:text-red-200">
            {error}
          </div>
        )}

        {/* Verses list */}
        <ScrollArea className="max-h-80 rounded-2xl border border-slate-100 dark:border-slate-800 bg-white/80 dark:bg-slate-950/70">
          <div className="p-4 space-y-4">
            {loading && (
              <div className="flex items-center justify-center py-6 text-sm text-slate-500 gap-2">
                <Loader2 className="w-4 h-4 animate-spin" />
                Loading verses…
              </div>
            )}

            {!loading && verses.length === 0 && (
              <div className="py-4 text-center text-sm text-slate-500">
                No verses loaded yet.
              </div>
            )}

            {!loading &&
              verses.map((v) => (
                <div
                  key={`${v.reference}-${v.version}`}
                  className="rounded-xl bg-slate-50 dark:bg-slate-900/80 px-4 py-3 space-y-1"
                >
                  <div className="text-xs font-semibold text-amber-700 dark:text-amber-300 flex items-center gap-1">
                    <ChevronDown className="w-3 h-3 opacity-70" />
                    {v.reference} • {v.version}
                  </div>
                  <div className="text-sm text-slate-800 dark:text-slate-100 leading-relaxed whitespace-pre-wrap">
                    {v.text || "(Verse text not available yet.)"}
                  </div>
                </div>
              ))}
          </div>
        </ScrollArea>

        {/* Reload button */}
        <div className="flex justify-end">
          <Button
            variant="outline"
            size="sm"
            onClick={() => selectedTopic && loadVerses(selectedTopic, selectedVersion)}
            disabled={loading}
          >
            {loading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
            Refresh verses
          </Button>
        </div>
      </CardContent>
    </Card>
  );
}


Then, on your Home page (or wherever that tile lives), just render:

    <div className="p-4 space-y-6">
      {/* other tiles ... */}
      <TopicalBibleSearchCard />
      {/* ... */}
    </div>
  );
}

// Expected response shape for each verse
GET /api/bible-verse?reference=John%203:16&version=KJV

// should return JSON like:
{
  "reference": "John 3:16",
  "text": "For God so loved the world, ...", // from the chosen version
  "version": "KJV"
}
