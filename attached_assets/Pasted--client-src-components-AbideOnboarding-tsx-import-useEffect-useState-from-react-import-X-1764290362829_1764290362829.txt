// client/src/components/AbideOnboarding.tsx
import { useEffect, useState } from "react";
import { X, ArrowRight } from "lucide-react";
import {
  isAbideOnboardingCompleted,
  markAbideOnboardingCompleted,
} from "@/lib/abideStorage";
import { tapHaptic } from "@/lib/nativeEnhancements";

const slides = [
  {
    title: "Welcome to ABIDE",
    text: "ABIDE is your daily walk with Jesus where spiritual growth becomes visible.",
  },
  {
    title: "Hear · Reflect · Act · Pray",
    text: "Each day, you’ll hear God’s Word, reflect with Him, choose one simple act of obedience, and pray it in.",
  },
  {
    title: "See Your Fruit Grow",
    text: "As you walk with Jesus, the fruit of the Spirit will grow on your Tree — love, joy, peace, and more.",
  },
];

export default function AbideOnboarding() {
  const [visible, setVisible] = useState(false);
  const [index, setIndex] = useState(0);

  useEffect(() => {
    if (!isAbideOnboardingCompleted()) {
      setVisible(true);
    }
  }, []);

  if (!visible) return null;

  const slide = slides[index];

  const handleNext = async () => {
    await tapHaptic();
    if (index < slides.length - 1) {
      setIndex((i) => i + 1);
    } else {
      markAbideOnboardingCompleted();
      setVisible(false);
    }
  };

  const handleSkip = async () => {
    await tapHaptic();
    markAbideOnboardingCompleted();
    setVisible(false);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 px-6">
      <div className="relative w-full max-w-sm rounded-3xl bg-gradient-to-br from-slate-950 to-slate-900 border border-slate-700/70 px-5 pt-5 pb-4 shadow-xl shadow-black/80">
        <button
          onClick={handleSkip}
          className="absolute top-3 right-3 text-slate-400"
        >
          <X className="h-4 w-4" />
        </button>

        <p className="text-xs uppercase tracking-[0.2em] text-emerald-300/80 mb-2">
          Abide
        </p>
        <h2 className="text-[18px] font-semibold text-slate-50 mb-2">
          {slide.title}
        </h2>
        <p className="text-[13px] text-slate-200 mb-4">{slide.text}</p>

        <div className="flex items-center justify-between mt-2">
          <div className="flex gap-1">
            {slides.map((_, i) => (
              <div
                key={i}
                className={`h-1.5 w-6 rounded-full ${
                  i === index ? "bg-emerald-400" : "bg-slate-700"
                }`}
              />
            ))}
          </div>

          <button
            onClick={handleNext}
            className="inline-flex items-center rounded-full bg-emerald-400 px-4 py-1.5 text-[12px] font-semibold text-black"
          >
            {index === slides.length - 1 ? "Start Abiding" : "Next"}
            <ArrowRight className="h-3 w-3 ml-1" />
          </button>
        </div>

        <p className="mt-3 text-[11px] text-slate-400">
          *ABIDE — where spiritual growth becomes visible.*
        </p>
      </div>
    </div>
  );
}
