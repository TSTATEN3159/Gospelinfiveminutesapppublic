## 1. Frontend service – make sure we actually call the Trivia API correctly

**File:** `client/src/services/triviaService.ts`

Replace the whole file with this (or adapt it if you already export a `getBibleTrivia` function):

```ts
// client/src/services/triviaService.ts

export type TriviaLevel = 'beginner' | 'intermediate' | 'advanced';

export interface RawTriviaQuestion {
  id?: string;
  question?: string;
  text?: string;
  prompt?: string;
  options?: string[];
  choices?: string[];
  correctIndex?: number;
  answerIndex?: number;
  correctOptionIndex?: number;
  reference?: string;
  verseReference?: string;
  scripture?: string;
}

export interface TriviaQuestion {
  id: string;
  text: string;
  choices: string[];
  correctIndex: number;
  reference?: string;
}

/**
 * Normalizes whatever shape the backend returns into a clean TriviaQuestion
 */
function normalizeQuestion(raw: RawTriviaQuestion, index: number): TriviaQuestion {
  const text =
    raw.question ??
    raw.text ??
    raw.prompt ??
    'Question missing from API response';

  const choices = raw.choices ?? raw.options ?? [];

  const correctIndex =
    raw.correctIndex ??
    raw.answerIndex ??
    raw.correctOptionIndex ??
    0;

  const reference =
    raw.reference ??
    raw.verseReference ??
    raw.scripture ??
    undefined;

  return {
    id: raw.id ?? String(index),
    text,
    choices,
    correctIndex,
    reference,
  };
}

/**
 * Fetches one round of trivia (10 questions) from your backend.
 * IMPORTANT: useAI: true ensures we use the AI generator and not a dummy list.
 */
export async function fetchBibleTriviaQuestions(
  level: TriviaLevel,
  count = 10
): Promise<TriviaQuestion[]> {
  const response = await fetch('/api/bible-trivia', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      level,
      count,
      useAI: true,
    }),
  });

  if (!response.ok) {
    throw new Error(`Failed to fetch bible trivia: ${response.statusText}`);
  }

  const data = await response.json();

  // Support either { questions: [...] } or just [...] from backend
  const rawList: RawTriviaQuestion[] = Array.isArray(data.questions)
    ? data.questions
    : Array.isArray(data)
    ? data
    : [];

  return rawList.map(normalizeQuestion);
}
```

This fixes two big things:

1. **Forces `useAI: true`** so we don’t silently hit some static/local question set.
2. **Normalizes any question shape** coming from the backend so you don’t have weird undefined property bugs.

              {!isFinalLevel && (

  // Question screen

        <CardHeader>
