1.1 Create a trivia store with RAM cache + JSON “DB”

New file: server/triviaStore.ts

// server/triviaStore.ts
import fs from "fs";
import path from "path";
import { generateBibleTriviaQuestions } from "./ai-trivia-generator";

export type TriviaLevel = "beginner" | "intermediate" | "advanced";

export type TriviaCategory =
  | "oldTestament"
  | "gospels"
  | "epistles"
  | "prophecy"
  | "peopleOfGod"
  | "geography";

export interface StoredTriviaQuestion {
  id: string;
  level: TriviaLevel;
  question: string;
  choices: string[];
  correctIndex: number;
  reference?: string;
  category?: TriviaCategory;
}

// Location for simple JSON “DB”
const DATA_DIR = path.join(__dirname, "..", "data");
const DATA_FILE = path.join(DATA_DIR, "bible-trivia.json");

// In-memory pool for **instant** access
const memoryPool: Record<TriviaLevel, StoredTriviaQuestion[]> = {
  beginner: [],
  intermediate: [],
  advanced: [],
};

// ---------- Disk helpers ----------

function ensureDataDir() {
  if (!fs.existsSync(DATA_DIR)) {
    fs.mkdirSync(DATA_DIR, { recursive: true });
  }
}

function loadFromDisk() {
  try {
    ensureDataDir();
    if (!fs.existsSync(DATA_FILE)) return;

    const raw = fs.readFileSync(DATA_FILE, "utf8");
    if (!raw) return;
    const parsed: StoredTriviaQuestion[] = JSON.parse(raw);

    // Rebuild memory pool
    (["beginner", "intermediate", "advanced"] as TriviaLevel[]).forEach((lvl) => {
      memoryPool[lvl] = parsed.filter((q) => q.level === lvl);
    });

    console.log("[TriviaStore] Loaded", parsed.length, "questions from disk");
  } catch (err) {
    console.error("[TriviaStore] Failed to load from disk:", err);
  }
}

function saveToDisk() {
  try {
    ensureDataDir();
    const all: StoredTriviaQuestion[] = [
      ...memoryPool.beginner,
      ...memoryPool.intermediate,
      ...memoryPool.advanced,
    ];
    fs.writeFileSync(DATA_FILE, JSON.stringify(all, null, 2), "utf8");
  } catch (err) {
    console.error("[TriviaStore] Failed to save to disk:", err);
  }
}

// Load once at server startup
loadFromDisk();

// ---------- Core logic ----------

// Normalize what AI returns into StoredTriviaQuestion[]
function normalizeFromAI(
  level: TriviaLevel,
  rawList: any[]
): StoredTriviaQuestion[] {
  return rawList.map((q, index) => {
    const text = q.question ?? q.text ?? q.prompt ?? "Missing question text";
    const choices = q.choices ?? q.options ?? [];
    const correctIndex =
      q.correctIndex ?? q.answerIndex ?? q.correctOptionIndex ?? 0;
    const reference = q.reference ?? q.verseReference ?? q.scripture ?? undefined;

    const category: TriviaCategory | undefined = q.category;

    return {
      id: q.id ?? `${level}-${Date.now()}-${index}`,
      level,
      question: text,
      choices,
      correctIndex,
      reference,
      category,
    };
  });
}

// Fill the pool for a level up to targetSize
export async function topUpTriviaPool(
  level: TriviaLevel,
  targetSize = 200
): Promise<void> {
  const pool = memoryPool[level];

  if (pool.length >= targetSize) return;

  const needed = targetSize - pool.length;
  // Generate in chunks of 10 to keep calls small
  const batchSize = Math.min(10, needed);

  console.log(`[TriviaStore] Topping up ${level} by ${batchSize} questions...`);

  const aiQuestions = await generateBibleTriviaQuestions({
    level,
    count: batchSize,
    useAI: true,
  });

  const normalized = normalizeFromAI(level, aiQuestions);
  pool.push(...normalized);
  saveToDisk();

  console.log(
    `[TriviaStore] ${level} pool now has ${pool.length} questions (target ${targetSize})`
  );
}

// Simple random sample (without removing, so questions can be reused)
function sampleWithoutReplacement<T>(arr: T[], count: number): T[] {
  if (arr.length <= count) return [...arr];

  const indices = Array.from({ length: arr.length }, (_, i) => i);
  for (let i = indices.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [indices[i], indices[j]] = [indices[j], indices[i]];
  }
  return indices.slice(0, count).map((i) => arr[i]);
}

// Main function used by your API route
export async function getRandomTriviaQuestions(
  level: TriviaLevel,
  count: number
): Promise<StoredTriviaQuestion[]> {
  const pool = memoryPool[level];

  // If not enough in memory, synchronously top up at least what we need
  if (pool.length < count) {
    await topUpTriviaPool(level, count * 5); // make a small buffer too
  }

  // By now, we should have enough
  const selected = sampleWithoutReplacement(memoryPool[level], count);

  // Fire-and-forget: keep topping up in background toward larger buffer
  topUpTriviaPool(level, 200).catch((err) =>
    console.error("[TriviaStore] background topUp error:", err)
  );

  return selected;
}


This file gives you:

RAM cache → answer in milliseconds

Disk “DB” → questions persist across server restarts

Background top-up → new questions trickle in without users waiting

1.2 Change the trivia API route to use the pool

In server/routes.ts, find your /api/bible-trivia route and replace its inner logic with this:

// server/routes.ts
import { z } from "zod";
import {
  getRandomTriviaQuestions,
  TriviaLevel,
} from "./triviaStore";

// ...

const bibleTriviaSchema = z.object({
  level: z.enum(["beginner", "intermediate", "advanced"]).default("beginner"),
  count: z.number().min(1).max(10).default(10),
  useAI: z.boolean().default(true), // kept for compatibility but no longer required every time
});

app.post("/api/bible-trivia", async (req, res) => {
  try {
    const { level, count } = bibleTriviaSchema.parse(req.body);

    const typedLevel = level as TriviaLevel;
    const questions = await getRandomTriviaQuestions(typedLevel, count);

    // Return in the shape the frontend expects
    res.json({
      questions: questions.map((q) => ({
        id: q.id,
        question: q.question,
        options: q.choices,
        correctIndex: q.correctIndex,
        reference: q.reference,
        category: q.category,
      })),
    });
  } catch (err: any) {
    console.error("Bible trivia error:", err);
    res.status(500).json({ error: "Failed to get trivia questions." });
  }
});


Now your API:

Never waits for OpenAI unless the pool is nearly empty

Returns questions from RAM most of the time → huge speed boost
