# ====== 5) Create a tiny helper you can import in your app (src/liveUpdates.ts) ======
# - Default behavior: safe "Background" updater (no interruptions)
# - Also includes an opt-in 'always-latest' flow guarded by a reload-block flag
mkdir -p src
cat > src/liveUpdates.ts <<'TS'
import { App } from '@capacitor/app';
import * as LiveUpdates from '@capacitor/live-updates';

/**
 * Call this once during app startup (e.g., in main.ts / main.tsx).
 * Default: background mode (no extra code needed).
 * If you switch to manual mode (autoUpdateMethod: 'none'), this enables the "Always Latest—with guardrails" approach.
 */
export async function initializeLiveUpdates(manualMode = false) {
  if (!manualMode) return; // In background mode, SDK handles checks; nothing else to do.

  // Manual mode ("Always Latest"): check once on load, then on each resume.
  const syncAndFlag = async () => {
    const r = await LiveUpdates.sync();
    localStorage.shouldReloadApp = String(r.activeApplicationPathChanged);
  };

  await syncAndFlag();

  App.addListener('resume', async () => {
    const shouldReload = localStorage.shouldReloadApp === 'true';
    const block = localStorage.shouldBlockReload === 'true';
    if (shouldReload && !block) {
      await LiveUpdates.reload();
    } else {
      await syncAndFlag();
    }
  });
}

/** Set to true during sensitive flows (login/purchase), false afterwards */
export function setReloadBlocked(block: boolean) {
  localStorage.shouldBlockReload = String(block);
}
TS

# ====== 6) Try to auto-wire the helper into your app entry (best effort) ======
if [ -f src/main.tsx ]; then
  if ! grep -q "initializeLiveUpdates" src/main.tsx; then
    echo "Patching src/main.tsx to call initializeLiveUpdates(false) (background mode)…"
    # add import at top if not present
    sed -i '1i import { initializeLiveUpdates } from "./liveUpdates";' src/main.tsx
    # call it near the top (safe no-op if background mode)
    sed -i '1a initializeLiveUpdates(false);' src/main.tsx
  fi
elif [ -f src/main.ts ]; then
  if ! grep -q "initializeLiveUpdates" src/main.ts; then
    echo "Patching src/main.ts to call initializeLiveUpdates(false) (background mode)…"
    sed -i '1i import { initializeLiveUpdates } from "./liveUpdates";' src/main.ts
    sed -i '1a initializeLiveUpdates(false);' src/main.ts
  fi
else
  echo "No src/main.tsx or src/main.ts found. You can import initializeLiveUpdates(false) in your app entry."
fi
