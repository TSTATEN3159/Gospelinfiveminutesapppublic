// client/src/hooks/useImageGenerator.ts
import { useState, useCallback, useRef } from 'react';
import { getBackgroundById } from '@/config/backgroundImages';

export interface ImageSettings {
  verseText: string;
  verseReference: string;
  backgroundId: string;
  textPosition: 'top' | 'center' | 'bottom';
  textColor: string;
  fontSize: number;       // Treated as a *max* font size
  fontFamily: string;
  textShadow: boolean;
  textAlign: 'left' | 'center' | 'right';  // NEW
  showTextPanel: boolean;                  // NEW
}

export interface GeneratedImage {
  dataUrl: string;
  blob: Blob;
  settings: ImageSettings;
}

const DEFAULT_CANVAS_WIDTH = 1080;
const DEFAULT_CANVAS_HEIGHT = 1080;

export function useImageGenerator() {
  const [isGenerating, setIsGenerating] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  const loadImage = useCallback((src: string): Promise<HTMLImageElement> => {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.crossOrigin = 'anonymous';
      img.onload = () => resolve(img);
      img.onerror = () => reject(new Error(`Failed to load image: ${src}`));
      img.src = src;
    });
  }, []);

  const wrapText = useCallback((
    ctx: CanvasRenderingContext2D,
    text: string,
    maxWidth: number
  ): string[] => {
    const paragraphs = text.split('\n');
    const lines: string[] = [];

    for (const paragraph of paragraphs) {
      const words = paragraph.split(' ');
      let currentLine = '';

      for (const word of words) {
        const testLine = currentLine ? `${currentLine} ${word}` : word;
        const width = ctx.measureText(testLine).width;

        if (width > maxWidth && currentLine) {
          lines.push(currentLine);
          currentLine = word;
        } else {
          currentLine = testLine;
        }
      }

      if (currentLine) {
        lines.push(currentLine);
      }
    }

    return lines;
  }, []);

  const computeTextLayout = useCallback((
    ctx: CanvasRenderingContext2D,
    settings: ImageSettings,
    canvasWidth: number,
    canvasHeight: number
  ) => {
    const maxOuterFontSize = settings.fontSize || 32;
    const minFontSize = 14;

    const horizontalPadding = canvasWidth * 0.08;   // 8% left/right margin
    const verticalPadding = canvasHeight * 0.08;     // 8% top/bottom margin
    const maxTextWidth = canvasWidth - horizontalPadding * 2;
    const maxTextHeight = canvasHeight - verticalPadding * 2;

    let fontSize = maxOuterFontSize;
    let verseLines: string[] = [];
    let referenceLines: string[] = [];
    let totalHeight = 0;

    while (fontSize >= minFontSize) {
      ctx.font = `${fontSize}px "${settings.fontFamily}"`;
      verseLines = wrapText(ctx, settings.verseText, maxTextWidth);

      const refFontSize = Math.round(fontSize * 0.7);
      ctx.font = `italic ${refFontSize}px "${settings.fontFamily}"`;
      referenceLines = settings.verseReference
        ? wrapText(ctx, settings.verseReference, maxTextWidth)
        : [];

      totalHeight = 0;

      // verse block
      ctx.font = `${fontSize}px "${settings.fontFamily}"`;
      for (const line of verseLines) {
        const metrics = ctx.measureText(line);
        const lineHeight =
          (metrics.actualBoundingBoxAscent || fontSize * 0.8) +
          (metrics.actualBoundingBoxDescent || fontSize * 0.2);

        totalHeight += lineHeight * 1.2;
      }

      if (referenceLines.length > 0) {
        totalHeight += fontSize * 0.5;
      }

      // reference block
      ctx.font = `italic ${refFontSize}px "${settings.fontFamily}"`;
      for (const line of referenceLines) {
        const metrics = ctx.measureText(line);
        const lineHeight =
          (metrics.actualBoundingBoxAscent || refFontSize * 0.8) +
          (metrics.actualBoundingBoxDescent || refFontSize * 0.2);

        totalHeight += lineHeight * 1.2;
      }

      if (totalHeight <= maxTextHeight) {
        const textBoxX = (canvasWidth - maxTextWidth) / 2;

        return {
          fontSize,
          refFontSize,
          verseLines,
          referenceLines,
          totalHeight,
          maxTextWidth,
          horizontalPadding,
          verticalPadding,
          textBoxX,
        };
      }

      fontSize -= 1;
    }

    // Fallback with minimum font size
    const finalFontSize = minFontSize;
    ctx.font = `${finalFontSize}px "${settings.fontFamily}"`;
    verseLines = wrapText(ctx, settings.verseText, maxTextWidth);

    const refFs = Math.round(finalFontSize * 0.7);
    ctx.font = `italic ${refFs}px "${settings.fontFamily}"`;
    referenceLines = settings.verseReference
      ? wrapText(ctx, settings.verseReference, maxTextWidth)
      : [];

    totalHeight = 0;
    ctx.font = `${finalFontSize}px "${settings.fontFamily}"`;
    for (const line of verseLines) {
      const metrics = ctx.measureText(line);
      const lineHeight =
        (metrics.actualBoundingBoxAscent || finalFontSize * 0.8) +
        (metrics.actualBoundingBoxDescent || finalFontSize * 0.2);

      totalHeight += lineHeight * 1.2;
    }

    if (referenceLines.length > 0) {
      totalHeight += finalFontSize * 0.5;
    }

    ctx.font = `italic ${refFs}px "${settings.fontFamily}"`;
    for (const line of referenceLines) {
      const metrics = ctx.measureText(line);
      const lineHeight =
        (metrics.actualBoundingBoxAscent || refFs * 0.8) +
        (metrics.actualBoundingBoxDescent || refFs * 0.2);

      totalHeight += lineHeight * 1.2;
    }

    const textBoxX = (canvasWidth - maxTextWidth) / 2;

    return {
      fontSize: finalFontSize,
      refFontSize: refFs,
      verseLines,
      referenceLines,
      totalHeight,
      maxTextWidth,
      horizontalPadding,
      verticalPadding,
      textBoxX,
    };
  }, [wrapText]);

  const generateImage = useCallback(async (settings: ImageSettings): Promise<GeneratedImage> => {
    setIsGenerating(true);
    setError(null);

    try {
      const background = getBackgroundById(settings.backgroundId);
      if (!background) {
        throw new Error('Background not found');
      }

      const canvas = document.createElement('canvas');
      canvas.width = DEFAULT_CANVAS_WIDTH;
      canvas.height = DEFAULT_CANVAS_HEIGHT;

      const ctx = canvas.getContext('2d');
      if (!ctx) {
        throw new Error('Failed to get canvas context');
      }

      // --- Background ---
      if (background.type === 'solid-color' && background.color) {
        ctx.fillStyle = background.color;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
      } else if (background.type === 'image' && background.url) {
        const img = await loadImage(background.url);

        const scale = Math.max(
          canvas.width / img.width,
          canvas.height / img.height
        );
        const scaledWidth = img.width * scale;
        const scaledHeight = img.height * scale;
        const x = (canvas.width - scaledWidth) / 2;
        const y = (canvas.height - scaledHeight) / 2;

        ctx.drawImage(img, x, y, scaledWidth, scaledHeight);
      }

      // --- Layout ---
      const layout = computeTextLayout(ctx, settings, canvas.width, canvas.height);

      const {
        fontSize,
        refFontSize,
        verseLines,
        referenceLines,
        totalHeight,
        maxTextWidth,
        horizontalPadding,
        verticalPadding,
        textBoxX,
      } = layout;

      const safeTop = verticalPadding;
      const safeBottom = canvas.height - verticalPadding;

      let startY: number;
      switch (settings.textPosition) {
        case 'top':
          startY = safeTop;
          break;
        case 'bottom':
          startY = Math.max(safeTop, safeBottom - totalHeight);
          break;
        case 'center':
        default:
          startY = Math.max(safeTop, (canvas.height - totalHeight) / 2);
          break;
      }

      // --- Optional translucent text panel ---
      if (settings.showTextPanel) {
        const panelPaddingY = fontSize * 0.6;
        const panelX = horizontalPadding * 0.5;
        const panelWidth = canvas.width - panelX * 2;
        const panelY = Math.max(safeTop * 0.6, startY - panelPaddingY);
        const panelHeight = totalHeight + panelPaddingY * 2;

        ctx.save();
        ctx.fillStyle = 'rgba(0, 0, 0, 0.35)';
        ctx.fillRect(panelX, panelY, panelWidth, panelHeight);
        ctx.restore();
      }

      // --- Text styling ---
      ctx.fillStyle = settings.textColor;
      ctx.textBaseline = 'top';

      if (settings.textShadow) {
        ctx.shadowColor = 'rgba(0, 0, 0, 0.7)';
        ctx.shadowBlur = 8;
        ctx.shadowOffsetX = 2;
        ctx.shadowOffsetY = 2;
      } else {
        ctx.shadowColor = 'transparent';
        ctx.shadowBlur = 0;
        ctx.shadowOffsetX = 0;
        ctx.shadowOffsetY = 0;
      }

      // Horizontal alignment
      let textX: number;
      if (settings.textAlign === 'left') {
        ctx.textAlign = 'left';
        textX = textBoxX;
      } else if (settings.textAlign === 'right') {
        ctx.textAlign = 'right';
        textX = textBoxX + maxTextWidth;
      } else {
        ctx.textAlign = 'center';
        textX = canvas.width / 2;
      }

      // --- Draw verse ---
      ctx.font = `${fontSize}px "${settings.fontFamily}"`;
      let currentY = startY;

      for (const line of verseLines) {
        ctx.fillText(line, textX, currentY);
        const metrics = ctx.measureText(line);
        const lineHeight =
          (metrics.actualBoundingBoxAscent || fontSize * 0.8) +
          (metrics.actualBoundingBoxDescent || fontSize * 0.2);

        currentY += lineHeight * 1.2;
      }

      // --- Draw reference ---
      if (referenceLines.length > 0) {
        currentY += fontSize * 0.5;

        ctx.font = `italic ${refFontSize}px "${settings.fontFamily}"`;
        for (const line of referenceLines) {
          ctx.fillText(line, textX, currentY);
          const metrics = ctx.measureText(line);
          const lineHeight =
            (metrics.actualBoundingBoxAscent || refFontSize * 0.8) +
            (metrics.actualBoundingBoxDescent || refFontSize * 0.2);

          currentY += lineHeight * 1.2;
        }
      }

      const dataUrl = canvas.toDataURL('image/png', 0.95);

      const blob = await new Promise<Blob>((resolve, reject) => {
        canvas.toBlob(
          (b) => {
            if (b) resolve(b);
            else reject(new Error('Failed to create blob'));
          },
          'image/png',
          0.95
        );
      });

      canvasRef.current = canvas;

      return { dataUrl, blob, settings };
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Failed to generate image';
      setError(message);
      throw err;
    } finally {
      setIsGenerating(false);
    }
  }, [computeTextLayout, loadImage]);

  return {
    generateImage,
    isGenerating,
    error,
    canvasRef: canvasRef.current,
  };
}

What you should see after these changes
When you choose Top, the entire verse block hugs the upper safe margin (with padding), not floating in the center.
Long Scripture passages auto-shrink to fit gracefully inside the 1080×1080 image with margins.
The verse never runs off the edges or collapses into a weird single line.
You can create:
Centered devotional cards (center / center / panel ON).
Left-aligned teaching cards (top / left / panel ON).
Minimalist designs (any alignment, panel OFF, maybe no shadow).
