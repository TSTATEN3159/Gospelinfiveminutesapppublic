🎨 Tile Component Code Update (So They Rotate & Stay Unique)
// Random background picker per feature type
const pickBackground = (type: "devotional" | "plainMeaning") => {
  const images = {
    devotional: [
      "/images/tiles/devotional/devotional-1.jpg",
      "/images/tiles/devotional/devotional-2.jpg",
      "/images/tiles/devotional/devotional-3.jpg",
      "/images/tiles/devotional/devotional-4.jpg",
      "/images/tiles/devotional/devotional-5.jpg",
    ],
    plainMeaning: [
      "/images/tiles/plain-meaning/plain-meaning-1.jpg",
      "/images/tiles/plain-meaning/plain-meaning-2.jpg",
      "/images/tiles/plain-meaning/plain-meaning-3.jpg",
      "/images/tiles/plain-meaning/plain-meaning-4.jpg",
      "/images/tiles/plain-meaning/plain-meaning-5.jpg",
    ],
  };

  const arr = images[type];
  return arr[Math.floor(Math.random() * arr.length)];
};

// Example Usage in Tile UI
const backgroundImage = pickBackground("devotional"); 
// or: pickBackground("plainMeaning");
