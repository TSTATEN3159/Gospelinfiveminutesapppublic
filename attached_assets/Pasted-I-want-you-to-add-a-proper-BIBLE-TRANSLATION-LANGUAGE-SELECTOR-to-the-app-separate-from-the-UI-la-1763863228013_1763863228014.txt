export interface BibleTranslation {
  id: BibleTranslationId;
  label: string;       // Shown in the UI

  apiCode: string;     // Code used for API/backend calls

  {
    id: 'niv',
    label: 'English – NIV',
    shortLabel: 'NIV',
    languageCode: 'en',
    apiCode: 'NIV',

  {
    id: 'kjv',
    label: 'English – KJV',
    shortLabel: 'KJV',
    languageCode: 'en',
    apiCode: 'KJV',
  },
  {
    id: 'es-rvr1960',
    label: 'Español – RVR1960',
    shortLabel: 'RVR1960',
    languageCode: 'es',
    apiCode: 'RVR1960',
  },
];

export const DEFAULT_BIBLE_TRANSLATION_ID: BibleTranslationId = 'niv';

export function getBibleTranslationById(id: BibleTranslationId): BibleTranslation {

    BIBLE_TRANSLATIONS.find((t) => t.id === id) ??
    BIBLE_TRANSLATIONS.find((t) => t.id === DEFAULT_BIBLE_TRANSLATION_ID)!
  );
}

📌 2️⃣ Add a Bible Translation Store / Context
━━━━━━━━━━━━━━━━━━
Create a new file:

client/src/state/useBibleTranslationStore.ts

Use Zustand or simple context. If Zustand is already used in the app, use it; otherwise, you can add Zustand.

If Zustand is already installed, use this:

import { create } from 'zustand';
import { BibleTranslationId, DEFAULT_BIBLE_TRANSLATION_ID } from '@/config/bibleTranslations';

interface BibleTranslationState {
  translationId: BibleTranslationId;
  setTranslationId: (id: BibleTranslationId) => void;
}

const STORAGE_KEY = 'bible.translationId';

export const useBibleTranslationStore = create<BibleTranslationState>((set) => {
  let initial: BibleTranslationId = DEFAULT_BIBLE_TRANSLATION_ID;
  if (typeof window !== 'undefined') {
    const saved = window.localStorage.getItem(STORAGE_KEY) as BibleTranslationId | null;
    if (saved) {
      initial = saved;
    }
  }

  return {
    translationId: initial,
    setTranslationId: (id) => {
      if (typeof window !== 'undefined') {
        window.localStorage.setItem(STORAGE_KEY, id);
      }
      set({ translationId: id });
    },
  };
});

If Zustand is NOT used, you can either install it (npm install zustand) or implement an equivalent React Context pattern. Use whatever aligns with the existing state management approach.
