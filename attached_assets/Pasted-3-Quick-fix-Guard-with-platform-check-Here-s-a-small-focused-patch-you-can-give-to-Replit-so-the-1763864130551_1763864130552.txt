A) SHARE HANDLER

Find the share handler you wired earlier, something like:

const handleShare = async () => {
  try {
    await ShareCard.share({
      verseText: verse.text,
      reference: verse.ref,
    });
  } catch (e) {
    console.error('Share failed', e);
  }

Replace it with a platform-aware version:

const handleShare = async () => {
  try {
    const verseText = verse.text;
    const reference = verse.ref;
    const platform = Capacitor.getPlatform();

    if (Capacitor.isNativePlatform() && platform === 'ios') {
      // Use native share plugin (ShareCard or BrandedShareImage if present)
      if (ShareCard && typeof ShareCard.share === 'function') {
        await ShareCard.share({ verseText, reference });
      }
    } else {
      // Web or non-iOS fallback
      if (navigator.share) {
        await navigator.share({
          title: reference,
          text: verseText,
        });
      } else {
        console.log('Sharing is only fully supported in the iOS app.');
        // Optionally show a toast/snackbar here if one exists in the app
      }
    }
  } catch (e) {
    console.error('Share failed', e);
  }
};
