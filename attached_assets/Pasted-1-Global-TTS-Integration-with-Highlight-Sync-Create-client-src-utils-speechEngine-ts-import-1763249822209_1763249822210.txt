1️⃣ Global TTS Integration with Highlight Sync

📌 Create → client/src/utils/speechEngine.ts

import { TextToSpeech } from '@capacitor-community/text-to-speech';
import { getVoice } from '@/utils/voicePrefs';

let isSpeaking = false;

export async function speak(
  text: string,
  onWordHighlight?: (wordIndex: number) => void
) {
  if (!text) return;

  const selectedVoice = getVoice();

  const words = text.split(" ");
  let index = 0;

  isSpeaking = true;

  const interval = setInterval(() => {
    if (!isSpeaking || index >= words.length) {
      clearInterval(interval);
      return;
    }
    
    onWordHighlight?.(index);
    index++;
  }, 500);

  await TextToSpeech.speak({
    text,
    lang: "en-US",
    voice: selectedVoice,
    rate: 0.96,
    pitch: 1.03,
    volume: 1.0,
    category: "playback"
  });

  clearInterval(interval);
  isSpeaking = false;
}

export async function stopSpeech() {
  isSpeaking = false;
  await TextToSpeech.stop();
}

export function toggleSpeech(text: string, cb?: any) {
  if (isSpeaking) {
    stopSpeech();
  } else {
    speak(text, cb);
  }
}

2️⃣ User Voice Preferences Storage

📌 Create → client/src/utils/voicePrefs.ts

const VOICE_PREF_KEY = "preferredVoice";

export function setVoice(id: string) {
  localStorage.setItem(VOICE_PREF_KEY, id);
}

export function getVoice(): string {
  return localStorage.getItem(VOICE_PREF_KEY) || "";
}

3️⃣ Voice Settings Page under More tab

📌 Create → client/src/pages/VoiceSettingsPage.tsx

import { setVoice, getVoice } from "@/utils/voicePrefs";
import { Card } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { speak, stopSpeech } from "@/utils/speechEngine";
import { useEffect, useState } from "react";

const voices = [
  { id: "com.apple.ttsbundle.Siri_Female_en-US_premium", label: "Female — Warm US" },
  { id: "com.apple.ttsbundle.Siri_Male_en-US_premium", label: "Male — Bold US" },
  { id: "com.apple.ttsbundle.Alison_en-GB_premium", label: "British Female — Soft" },
  { id: "com.apple.ttsbundle.Arthur_en-GB_premium", label: "British Male — Classic" },
  { id: "", label: "System Default" },
];

export default function VoiceSettingsPage() {
  const [selected, setSelected] = useState(getVoice());

  return (
    <div className="p-4">
      <h1 className="text-xl font-bold mb-4">Voice Settings</h1>

      {voices.map(v => (
        <Card
          key={v.id}
          className={`p-4 mb-2 cursor-pointer border-2 ${
            selected === v.id ? "border-blue-600" : "border-gray-300"
          }`}
          onClick={() => {
            setSelected(v.id);
            setVoice(v.id);
            stopSpeech();
            speak("This is how I sound.");
          }}
        >
          {v.label}
        </Card>
      ))}

      <Button className="w-full mt-4">
        Done
      </Button>
    </div>
  );
}
