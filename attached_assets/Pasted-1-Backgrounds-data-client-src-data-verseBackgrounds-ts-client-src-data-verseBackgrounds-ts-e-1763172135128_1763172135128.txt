2️⃣ Background picker component – client/src/components/VerseBackgroundPicker.tsx
// client/src/components/VerseBackgroundPicker.tsx

import React from "react";
import { VERSE_BACKGROUNDS, VerseBackground } from "@/data/verseBackgrounds";

interface VerseBackgroundPickerProps {
  selectedId: string;
  onChange: (background: VerseBackground) => void;
}

export const VerseBackgroundPicker: React.FC<VerseBackgroundPickerProps> = ({
  selectedId,
  onChange,
}) => {
  return (
    <div className="w-full">
      <h3 className="text-sm font-semibold mb-2 text-slate-700 dark:text-slate-200">
        Choose Background
      </h3>
      <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
        {VERSE_BACKGROUNDS.map((bg) => {
          const isSelected = bg.id === selectedId;
          return (
            <button
              key={bg.id}
              type="button"
              onClick={() => onChange(bg)}
              className={`
                relative group rounded-lg overflow-hidden border
                ${isSelected ? "border-blue-500 ring-2 ring-blue-400" : "border-slate-200 dark:border-slate-700"}
              `}
            >
              <div className="aspect-[9/16] w-full overflow-hidden">
                <img
                  src={bg.thumbnailUrl || bg.imageUrl}
                  alt={bg.name}
                  className={`
                    w-full h-full object-cover transition-transform duration-300
                    group-hover:scale-105
                  `}
                />
                <div className="absolute inset-0 bg-black/25 group-hover:bg-black/35 transition-colors" />
              </div>
              <div className="absolute bottom-0 left-0 right-0 px-2 py-1 bg-black/50 text-[0.7rem] text-white text-center">
                {bg.name}
              </div>
            </button>
          );
        })}
      </div>
    </div>
  );
};
3️⃣ Example screen that wires everything together
This assumes you already have the VerseImageComposer component we discussed earlier.
Here’s a full example screen that:
Lets the user change font size
Lets the user pick one of the spiritual backgrounds
Renders the verse + reference centered and auto-fitted on the chosen image
// client/src/pages/VerseImageScreen.tsx

import React, { useState } from "react";
import { VerseImageComposer } from "@/components/VerseImageComposer";
import { VerseBackgroundPicker } from "@/components/VerseBackgroundPicker";
import { VERSE_BACKGROUNDS } from "@/data/verseBackgrounds";

export default function VerseImageScreen() {
  // Example verse – wire this to your real Verse of the Day state
  const [verseText] = useState<string>(
    "The LORD is my shepherd; I shall not want."
  );
  const [reference] = useState<string>("Psalm 23:1");

  const [fontSize, setFontSize] = useState<number>(32);

  const [selectedBg, setSelectedBg] = useState(VERSE_BACKGROUNDS[0]);

  return (
    <div className="min-h-screen bg-slate-50 dark:bg-slate-900 px-4 py-6 pb-24">
      <div className="max-w-md mx-auto space-y-6">
        <h1 className="text-2xl font-bold text-slate-900 dark:text-slate-100 mb-2">
          Share Verse Image
        </h1>
        <p className="text-sm text-slate-600 dark:text-slate-300 mb-4">
          Choose a background and adjust the text size. The verse will
          automatically fit inside the image and stay centered.
        </p>

        {/* Background Picker */}
        <VerseBackgroundPicker
          selectedId={selectedBg.id}
          onChange={(bg) => setSelectedBg(bg)}
        />

        {/* Font size slider */}
        <div className="mt-4">
          <label className="text-sm font-semibold text-slate-700 dark:text-slate-200 flex justify-between mb-1">
            <span>Text Size</span>
            <span className="text-xs text-slate-500 dark:text-slate-400">
              {fontSize}px
            </span>
          </label>
          <input
            type="range"
            min={16}
            max={48}
            value={fontSize}
            onChange={(e) => setFontSize(Number(e.target.value))}
            className="w-full"
          />
        </div>

        {/* Verse image composer */}
        <VerseImageComposer
          verseText={verseText}
          reference={reference}
          backgroundUrl={selectedBg.imageUrl}
          userFontSize={fontSize}
        />
      </div>
    </div>
  );
}
