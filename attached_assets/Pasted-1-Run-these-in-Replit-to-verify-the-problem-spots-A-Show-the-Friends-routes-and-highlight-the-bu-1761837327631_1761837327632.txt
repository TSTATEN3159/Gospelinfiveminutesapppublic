2) Quick backend patch (fixes “Find Friends”)

Problem:
Your snippet shows:

  res.json({ success: true, friends: ... });

Fix: Read from req.params, validate, and harden errors.

// server/routes.ts  — replace the existing handler with this one
app.get("/api/friends/:userId", async (req, res) => {
  try {
    const { userId } = req.params;
    if (!userId) {
      return res.status(400).json({ success: false, error: "Missing userId param" });
    }
    const friends = await storage.getFriends(userId);
    return res.json({
      success: true,
      friends: friends.map(f => ({
        id: f.id,
        firstName: f.firstName,
        lastName: f.lastName,
        email: f.email,
        joinDate: f.joinDate
      }))
    });
  } catch (err: any) {
    console.error("[GET /api/friends/:userId] error:", err?.message || err);
    return res.status(500).json({ success: false, error: "Failed to fetch friends" });
  }
});

B) Frontend: request permissions explicitly before fetching

In ImportFriendsDialog.tsx (or wherever you tap “Import”), make sure you do this:

import { Contacts } from '@capacitor-community/contacts';
import { Capacitor } from '@capacitor/core';

async function importDeviceContacts(userId: string) {
  // iOS: request permission explicitly
  if (Capacitor.getPlatform() === 'ios') {
    const perm = await Contacts.requestPermissions();
    // statuses: 'granted' | 'denied' | 'prompt' | 'limited' (iOS 17+)
    if (!perm?.granted) {
      throw new Error('Contacts permission not granted');
    }
  }

  const res = await Contacts.getContacts({
    projection: {
      name: true,
      phones: true,
      emails: true
    }
  });

  const contacts = (res?.contacts || [])
    .map(c => ({
      firstName: c.name?.given ?? '',
      lastName:  c.name?.family ?? '',
      email: (c.emails?.[0]?.address ?? '').trim() || undefined,
      phone: (c.phones?.[0]?.number ?? '').trim() || undefined
    }))
    // require name + (email OR phone)
    .filter(c => (c.firstName || c.lastName) && (c.email || c.phone));

  // Limit to 50 during signup (matches your backend)
  const payload = { contacts: contacts.slice(0, 50) };

  const resp = await fetch(apiUrl(`/api/contacts/${userId}/import?fromSignup=true`), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });

  if (!resp.ok) {
    const txt = await resp.text();
    throw new Error(`Import failed: ${resp.status} ${txt}`);
  }

  return await resp.json();
}

You’re actually passing a real userId into /api/contacts/:userId/import. (If you’re anonymous, create/register first and persist userId.)

4) Quick backend hardening for contacts import (optional, but recommended)

Add validation so bad payloads don’t 500 your server:

app.post("/api/contacts/:userId/import", async (req, res) => {
  try {
    const { userId } = req.params;
    const fromSignup = req.query.fromSignup === 'true';

    if (!userId) return res.status(400).json({ success: false, error: "Missing userId" });
    if (!Array.isArray(req.body?.contacts)) {
      return res.status(400).json({ success: false, error: "contacts must be an array" });
    }

    const limit = fromSignup ? 50 : 1000;
    const contacts = req.body.contacts
      .slice(0, limit)
      .filter((c: any) => (c?.firstName || c?.lastName) && (c?.email || c?.phone));

    const importedContacts = await storage.importContacts(userId, contacts);
    const appUserContacts = await storage.findAppUsersFromContacts(userId);

    return res.json({
      success: true,
      message: `${importedContacts.length} contacts imported, ${appUserContacts.length} app users found`,
      totalImported: importedContacts.length,
      appUsersFound: appUserContacts.length
    });
  } catch (err: any) {
    console.error("[POST /api/contacts/:userId/import] error:", err?.message || err);
    return res.status(500).json({ success: false, error: "Failed to import contacts" });
  }
});
